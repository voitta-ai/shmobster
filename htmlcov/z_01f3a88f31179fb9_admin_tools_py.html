142def _held_answer(kind, key, channel, queue): 

143 """What a consumer that failed to take a request is told. Held and absent 

144 are different truths (#105): "no pending request" for a command another 

145 surface is running right now is exactly the contradiction the in-flight 

146 hold exists to prevent.""" 

147 state, _req = queue.status(key, channel) 

148 if state == "held": 

149 retval = ( 

150 f"[{key}] is already being acted on by another surface (a click or a " 

151 f"text approval got there first) -- do not retry; that surface will " 

152 f"report the outcome in this channel." 

153 ) 

154 return retval 

155 retval = f"no pending {kind} '{key}' in this channel." 

156 return retval

168def deny(request_id, ctx): 

169 """Drop a parked request without running it (#50 -- the Deny button). 

170 Trust-gated like approve_command; denial is a privileged act too, since a 

171 stranger could otherwise cancel work a trusted user asked for.""" 

172 if not is_trusted(ctx.get("user_id")): 

173 retval = _refuse(ctx, "deny a mutating command") 

174 return retval 

175 channel = ctx.get("channel") 

176 key = approvals.canonical(request_id) 

177 req = approvals.acquire(key, channel) 

178 if req is None: 

179 retval = _held_answer("request", key, channel, approvals) 

180 return retval 

181 try: 

182 retval = run_denied(key, req, ctx) 

183 finally: 

184 approvals.release(key) 

185 return retval

195def _post_alert(ctx, alert, card=None): 

196 """Post an alert straight to the channel, so the trusted-user tag is 

197 guaranteed rather than left to the model to relay. Returns whether Slack 

198 took it: refuse_click alerts at most once per user per request, so it has 

199 to know whether the one it is allowed to send actually landed. 

200 

201 `card` rides along as blocks when the alert is about something still 

202 actionable (#215), so the alert IS the affordance rather than a pointer to 

203 one. The alert text becomes the first block: blocks win over `text` in the 

204 client, so a card without it would drop the sentence explaining why the 

205 card is there, and `text` stays as the notification fallback.""" 

206 surface, channel = ctx.get("surface"), ctx.get("channel") 

207 if not (surface and channel): 

208 retval = False 

209 return retval 

210 blocks = None 

211 if card: 

212 blocks = [{"type": "section", "text": {"type": "mrkdwn", "text": alert}}] + list(card) 

213 retval = surface.post(channel, ctx.get("thread_ts"), alert, blocks) 

214 return retval

217def _refuse(ctx, what): 

218 """Loud refusal, and tag all trusted users so they know. 

219 

220 Deliberately does NOT assert the user asked for this (#59): the attempt to 

221 {what} may be the agent's own initiative during this user's turn, not a 

222 request from them. The old wording ("<user> asked me to change my config") 

223 read as a prompt-injection attack whenever the model self-initiated a 

224 set_policy call, and sent agents into false-alarm paralysis. 

225 

226 The button path does know who acted and writes its own alert -- see 

227 refuse_click, where the hedge above would be a lie (#94).""" 

228 alert = ( 

229 f":warning: A privileged change was attempted during {identity.mention(ctx.get('user_id'))}'s turn " 

230 f"(to {what}) and refused -- only trusted users may. This can be my own " 

231 f"doing, not necessarily their request. {_trusted_tags()} for visibility." 

232 ) 

233 _post_alert(ctx, alert) 

234 return _REFUSED

246def _alerted(request_id, channel, user_id): 

247 """Whether this user's click on this request here has already been alerted. 

248 

249 Leaving the card standing (#94) also leaves its buttons re-clickable, and 

250 the alert tags every trusted user -- so without this, one stranger holding 

251 down a button is an unbounded ping. The destructive chat_update used to 

252 swallow the second click by accident; this does it on purpose. In-memory 

253 and capped, like the queue it shadows. 

254 

255 Keyed by channel too (#107), and on the boot-unique queue key rather than 

256 the short id (#109): without either, a stale click on [1] silences the 

257 alert for a live [1] -- in another channel, or after a restart in the same 

258 one -- hiding exactly the unauthorized click trusted users are meant to 

259 hear about.""" 

260 retval = (str(request_id), channel, user_id) in _CLICK_ALERTED 

261 return retval

264def _mark_alerted(request_id, channel, user_id): 

265 """Record a *delivered* alert. Only delivery counts: a swallowed Slack 

266 failure on the one alert a user gets would otherwise mean the trusted users 

267 are never told and every retry is suppressed as already-told.""" 

268 _CLICK_ALERTED[(str(request_id), channel, user_id)] = None 

269 while len(_CLICK_ALERTED) > _CLICK_ALERT_MAX: 

270 del _CLICK_ALERTED[next(iter(_CLICK_ALERTED))]

273# Slack refuses a section whose text runs past 3000 characters, and a parked 

274# command can be longer than that (#215). The approval card now cuts such a 

275# command to a head and a tail with a link, so it fits, and a card that still 

276# fails to post withdraws its request; this stays as the guard, because an 

277# oversized card sent here would cost the alert itself: the post raises, `_post_alert` returns False, 

278# `_mark_alerted` never fires, and the trusted users are simply not told that 

279# an unauthorized click happened. That is the one thing this path exists to 

280# guarantee. So an oversized card degrades to the wording it replaced, which 

281# was always text and always fit. 

282_SLACK_SECTION_MAX = 3000 

283

285def _fits(card): 

286 """Whether every section in a card is inside Slack's per-section limit. 

287 

288 Checked rather than remembered: a size rule that lives only in a comment is 

289 one a longer command walks straight past, and the failure is silent at the 

290 surface that matters.""" 

291 retval = all( 

292 len(b.get("text", {}).get("text", "")) <= _SLACK_SECTION_MAX 

293 for b in (card or []) if b.get("type") == "section" 

294 ) 

295 return retval

298def _live_card(queue, key, req): 

299 """The still-parked request, rendered fresh and clickable (#215). 

300 

301 Each queue renders its own surface, so the renderer is chosen by queue 

302 rather than by sniffing the dict -- the same split `_QUEUES` already makes 

303 for `status`. Returns None when there is nothing to render, so a caller 

304 cannot turn a missing request into an empty card.""" 

305 if req is None: 

306 retval = None 

307 elif queue is proposals: 

308 retval = slack_blocks.proposal(key, req, _trusted_tags()) 

309 else: 

310 retval = slack_blocks.approval(key, req) 

311 return retval

314def refuse_click(request_id, ctx, action_id): 

315 """A non-trusted user pressed Approve or Deny on a parked command (#94). 

316 

317 Split from the model path because the two know different things. A button 

318 press has an unambiguous actor -- the model cannot click -- so the #59 

319 hedge would be a falsehood here, and saying it sent an agent hunting a 

320 self-initiated mutation that had never happened. 

321 

322 It also names the command, because the click is exactly when someone asks 

323 "what was parked?", and the card that answers that is the only other place 

324 it is written down. The queue is left untouched: refusing a click must not 

325 consume the request. 

326 """ 

327 label = _CLICK_LABELS.get(action_id, "a button") 

328 user_id = ctx.get("user_id") 

329 who = identity.mention(user_id) 

330 channel = ctx.get("channel") 

331 # Keyed on the canonical id, so a `#4` and a `4` are one request and one 

332 # alert -- and so a previous boot's id and this boot's cannot share an entry 

333 # (#109), which is what let a stale click silence the alert for a live one. 

334 key = approvals.canonical(request_id) 

335 if _alerted(key, channel, user_id): 

336 retval = "REFUSED: requester is not a trusted user. Trusted users have already been notified." 

337 return retval 

338 queue = _QUEUES.get(action_id, approvals) 

339 _state, req = queue.status(key, channel) 

340 # Kept before the flattening below, because rendering a live card needs the 

341 # request as its own queue's renderer expects it (#215). 

342 live = req 

343 # Only the pending branch sets one. A held request is already being acted 

344 # on and an absent one cannot be acted on at all, so a card in either would 

345 # be a button that does nothing -- the failure this is fixing, inverted. 

346 card = None 

347 if req is not None and "command" not in req: 

348 # A skill proposal (#129) renders as its name, not a command line. 

349 req = {"command": f"skill proposal `{req.get('name')}`"} 

350 if _state == "held": 

351 # Asked FIRST, because the queue goes quiet in the middle of this: a 

352 # trusted click acquires the request, rewrites the card to the claimed 

353 # button-less state, and the approve path pops it before running the 

354 # command -- so peek() says "pending" early in that window and "gone" 

355 # for the whole run, while neither is the useful answer. Only the hold 

356 # spans it. Promising live buttons here would send someone to press 

357 # buttons that are no longer there; calling it gone is worse still, 

358 # since the command is executing as the message is written. 

359 alert = ( 

360 f":warning: {who} clicked *{label}* on request [{key}], but only " 

361 f"trusted users may act on a parked command -- nothing ran. A trusted " 

362 f"user is already acting on it. {_trusted_tags()} for visibility." 

363 ) 

364 if req is not None: 

365 alert += "\n" + redact.scrub(f"```{req['command']}```") 

366 elif _state == "absent": 

367 # A stale card -- already claimed, denied, evicted, or cleared by a 

368 # restart. Saying "still parked" here would be the same kind of 

369 # confident falsehood this whole change exists to stop telling. 

370 alert = ( 

371 f":warning: {who} clicked *{label}* on request [{key}], which is " 

372 f"no longer pending in this channel -- nothing ran. Only trusted users " 

373 f"may act on a parked command. {_trusted_tags()} for visibility." 

374 ) 

375 else: 

376 # Says what happens next, not just what the rule is (#107). The card 

377 # and its buttons are deliberately left standing -- refusing a click 

378 # must not consume the request -- and without being told so the thread 

379 # reads this as the click having consumed something. 

380 # 

381 # It used to say the buttons "on the card above" were still live. True, 

382 # and useless: "above" is however far the conversation has travelled 

383 # since. Measured -- a click landed 43 minutes and a dozen messages 

384 # after the card was posted, the alert pointed up at it, and the 

385 # trusted user asked the agent to list what was parked and approved by 

386 # typing ids instead. The buttons worked the entire time; nobody could 

387 # find them. So the alert carries its own (#215), and the click lands 

388 # where the conversation is. 

389 # 

390 # Two cards for one request is fine: both resolve the same id, whichever 

391 # is clicked gets rewritten to the claimed state, and a click on the 

392 # other then reads "no longer pending", which is true. The command is 

393 # not repeated here -- the card renders it, scrubbed, and printing it 

394 # twice in one message is how a credential gets two chances (#72). 

395 card = _live_card(queue, key, live) 

396 if not _fits(card): 

397 card = None 

398 # Reached only for "pending", which `status()` never returns without a 

399 # request -- said out loud because the fallback below quotes it. 

400 shown = req["command"] if req is not None else f"(request {key})" 

401 head = (f":warning: {who} clicked *{label}* on request [{key}], but only " 

402 f"trusted users may act on a parked command -- nothing ran. The " 

403 f"request is still parked") 

404 if card is None: 

405 # No card to carry, so say where the original is and name the 

406 # command, exactly as before. Scrubbed like every other rendering of 

407 # a parked command (#72): a credential rides argv routinely, and 

408 # this is a fresh channel post. 

409 alert = ( 

410 f"{head} and the *Approve* / *Deny* buttons on the card above " 

411 f"are still live, so {_trusted_tags()} can act on it there." 

412 "\n" + redact.scrub(f"```{shown}```") 

413 ) 

414 else: 

415 alert = f"{head}, so {_trusted_tags()} can act on it right here." 

416 if _post_alert(ctx, alert, card): 

417 _mark_alerted(key, channel, user_id) 

418 retval = _REFUSED 

419 return retval

434def run_approved(request_id, req, ctx): 

435 """Core of Approve for a request the caller already acquired (#105): 

436 consume it (finish before execute -- the same pop-then-run rule as 

437 always), run it under the channel policy, report.""" 

438 approvals.finish(request_id) 

439 payload = req.get("payload") 

440 if payload and payload.get("kind") == "mcp": 

441 # A non-shell approval (#299): route on the payload kind rather than 

442 # running the display string as a command. mcp.run_payload re-reads the 

443 # server's credentials from the live policy; they never lived in req. 

444 out = mcp.run_payload(payload, ctx.get("channel")) 

445 else: 

446 policy = policy_mod.resolve(ctx.get("channel")) 

447 out = tools.execute(req["command"], policy) 

448 retval = f"APPROVED by {identity.mention(ctx.get('user_id'))} and ran: {req['command']}\n{out}" 

449 return retval

452def _approve_command(args, ctx): 

453 channel = ctx.get("channel") 

454 req_id = approvals.canonical(args.get("request_id", "")) 

455 req = approvals.acquire(req_id, channel) 

456 if req is None and approvals.status(req_id, channel)[0] == "held": 

457 retval = _held_answer("request", req_id, channel, approvals) 

458 return retval 

459 if req is None: 

460 # How many are parked, never which (#109). The likeliest way to reach 

461 # here is a trusted user quoting an id off a card from before a restart, 

462 # and this answer goes back into the tool loop as another turn the model 

463 # may act on -- so listing the live ids hands it an approvable id the 

464 # human never named, and "helpfully" retrying with one runs a command 

465 # nobody quoted. The count still says the queue is not empty, which is 

466 # what stops "no pending request" reading as "your request was 

467 # consumed" (#107); the id itself is on the card, where the human is. 

468 parked = len(approvals.ids(channel)) 

469 also = f" {parked} other request(s) are parked here." if parked else "" 

470 retval = ( 

471 f"no pending request '{req_id}' in this channel.{also} Do not guess " 

472 f"another id: ask the user to quote the id from the card they mean, " 

473 f"or to use its Approve button." 

474 ) 

475 return retval 

476 try: 

477 retval = run_approved(req_id, req, ctx) 

478 finally: 

479 approvals.release(req_id) 

480 return retval

483def dispatch(name, args, ctx): 

484 if name not in NAMES: 

485 return f"unknown admin tool: {name}" 

486 user_id = ctx.get("user_id") 

487 if not is_trusted(user_id): 

488 what = { 

489 "set_policy": "change my config restrictions", 

490 "reload_skills": "reload my skills", 

491 "propose_skill": "open a skill PR", 

492 "decline_skill": "decline a skill proposal", 

493 "deny_command": "deny a mutating command", 

494 }.get(name, "approve a mutating command") 

495 return _refuse(ctx, what) 

496 if name == "reload_skills": 

497 retval = _reload_skills() 

498 return retval 

499 if name == "approve_command": 

500 retval = _approve_command(args, ctx) 

501 return retval 

502 if name == "deny_command": 

503 retval = deny(args.get("request_id", ""), ctx) 

504 return retval 

505 if name == "propose_skill": 

506 retval = learning.propose(args.get("request_id", ""), ctx, scope=args.get("scope")) 

507 return retval 

508 if name == "decline_skill": 

509 retval = learning.decline(args.get("request_id", ""), ctx) 

510 return retval 

511 updates = { 

512 "cwd": args.get("cwd"), 

513 "github_repos": args.get("github_repos"), 

514 "aws_profile": args.get("aws_profile"), 

515 "exclude": args.get("exclude"), 

516 "skills": args.get("skills"), 

517 } 

518 channel_id = args.get("channel_id") or ctx.get("channel") 

519 # A linked gateway conversation has no policy of its own (#336). 

520 channel_id = policy_mod.linked(channel_id) or channel_id 

521 try: 

522 pol = config.set_channel_policy(channel_id, updates) 

523 return f"policy for {channel_id} updated (live): {pol}" 

524 except Exception as exc: 

525 return f"set_policy failed: {exc}"
