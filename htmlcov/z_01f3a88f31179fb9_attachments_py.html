1"""File attachments -> model content parts (#68); Slack's fetch lives here, the 

2gateway's in gateway_app (#413). 

3 

4Slack never puts an attachment in the message text; it arrives as 

5`event["files"]`, which the loop used to drop on the floor. So a mention like 

6"can you read this:" with an image under it reached the model as those four 

7words and nothing else, and the honest answer was "there is nothing attached". 

8 

9Two things about Slack file URLs are worth knowing before touching this: 

10 

11- They are not public. The bot token has to ride along as a bearer header, and 

12 fetching one needs the `files:read` scope. That token is workspace-wide, so 

13 it goes only to slack.com and its subdomains, and only for as long as the 

14 redirect chain stays there (#153). 

15- An unauthorized fetch does NOT fail. Slack answers **200 with the HTML 

16 sign-in page**, so a naive reader hands the model a login form and calls it a 

17 PNG. The content-type check below is what turns that into an error. 

18"""

35def _is_slack(url): 

36 """True for a URL whose host is slack.com or a subdomain of it. 

37 

38 The bearer is workspace-wide, so the question "may this URL have it" has to 

39 be asked of the URL rather than assumed from where it came: the download 

40 link arrives inside a Slack event, and an event is data.""" 

41 parts = urllib.parse.urlsplit(url) 

42 host = (parts.hostname or "").lower() 

43 # https only: a workspace-wide bearer does not travel in the clear, and 

44 # Slack does not serve these over http anyway. 

45 retval = parts.scheme == "https" and (host == "slack.com" or host.endswith(".slack.com")) 

46 return retval

49class _SlackOnlyRedirect(urllib.request.HTTPRedirectHandler): 

50 """urlopen follows redirects by default and copies the request headers into 

51 the next hop -- including Authorization, to whatever host the redirect 

52 names (confirmed on this box's Python 3.14: a 302 to another host received 

53 the bearer). Slack normally controls these URLs, so the likelihood is low 

54 and the credential is workspace-wide, which is the wrong pair of odds to 

55 accept for free. 

56 

57 Redirects within Slack still work, because a CDN hop is how a file download 

58 actually resolves; a redirect that leaves Slack fails instead of paying the 

59 token to whoever asked for it.""" 

60 

61 def redirect_request(self, req, fp, code, msg, headers, newurl): 

62 if not _is_slack(newurl): 

63 raise urllib.error.HTTPError( 

64 newurl, code, "redirect off slack.com; not sending the bot token", headers, fp 

65 ) 

66 retval = super().redirect_request(req, fp, code, msg, headers, newurl) 

67 return retval

73def _fetch(url): 

74 """Download one Slack-hosted file as bytes, or raise.""" 

75 if not _is_slack(url): 

76 raise ValueError(f"not a slack.com url: {urllib.parse.urlsplit(url).hostname!r}") 

77 req = urllib.request.Request( 

78 url, headers={"Authorization": f"Bearer {config.SLACK_BOT_TOKEN}"} 

79 ) 

80 with _OPENER.open(req, timeout=_TIMEOUT) as resp: 

81 ctype = (resp.headers.get("content-type") or "").lower() 

82 blob = resp.read(_MAX_BYTES + 1) 

83 if ctype.startswith("text/html"): 

84 # Slack serves the sign-in page with a 200, so this is the only signal 

85 # that the token was not accepted (usually: files:read not granted). 

86 raise ValueError("got Slack's sign-in page instead of the file (files:read granted?)") 

87 if len(blob) > _MAX_BYTES: 

88 raise ValueError(f"larger than the {_MAX_BYTES} byte cap") 

89 retval = blob 

90 return retval

93def _part(name, mime, blob): 

94 """One downloaded file as an OpenAI-style content part. LiteLLM translates 

95 `image_url` into each vendor's own image block, so a data: URI is the one 

96 shape that works across the waterfall.""" 

97 if mime.startswith("image/"): 

98 b64 = base64.b64encode(blob).decode() 

99 retval = { 

100 "type": "image_url", 

101 "image_url": {"url": f"data:{mime};base64,{b64}"}, 

102 } 

103 else: 

104 body = blob.decode("utf-8", "replace") 

105 retval = {"type": "text", "text": f"[attached file: {name}]\n{body}"} 

106 return retval

109# Under the channel cwd (#372). The directory ignores itself (a `*` .gitignore 

110# inside it), so a `git add -A` in a checkout never commits a client's file, 

111# and the repo's own .gitignore is never touched. 

112SAVE_DIR = ".attachments" 

113_ID_OK = re.compile(r"^[A-Za-z0-9]{1,64}$") 

114_NAME_BAD = re.compile(r"[^A-Za-z0-9._ -]") 

115_NAME_MAX = 200

118def _safe_name(name): 

119 """The Slack filename as one plain path component. The name is data from 

120 the event, so `../x`, `a/b` and a leading dot all come out as an ordinary 

121 file inside the attachment's own directory. A long name keeps its 

122 extension, so `cp`ing the file somewhere still yields a `.csv`.""" 

123 base = _NAME_BAD.sub("_", name).lstrip(". ").rstrip(". ") 

124 if len(base) > _NAME_MAX: 

125 stem, ext = os.path.splitext(base) 

126 ext = ext if len(ext) <= 16 else "" 

127 base = stem[:_NAME_MAX - len(ext)].rstrip(". ") + ext 

128 retval = base or "file" 

129 return retval

132def _dir_id(f, url): 

133 """The per-file directory: Slack's file id, or a hash of the url when the 

134 event carries no usable id.""" 

135 fid = str(f.get("id") or "") 

136 retval = fid if _ID_OK.match(fid) else "u" + hashlib.sha256(url.encode()).hexdigest()[:16] 

137 return retval

140def _open_dir(name, dir_fd): 

141 """Open (creating if needed) one directory under dir_fd without following 

142 a symlink. The channel can write its own cwd, so `.attachments` could be a 

143 link it planted; this process is not sandboxed, and following one would 

144 write wherever it pointed.""" 

145 try: 

146 os.mkdir(name, 0o700, dir_fd=dir_fd) 

147 except FileExistsError: 

148 pass 

149 retval = os.open(name, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=dir_fd) 

150 return retval

153def _save(cwd, sub, name, blob): 

154 """Write blob to <cwd>/.attachments/<sub>/<name>, byte-identical, and 

155 return the absolute path. A repeat save of the same file id rewrites the 

156 file atomically, so the path always holds Slack's bytes; an edit belongs 

157 on a copy.""" 

158 fds = [] 

159 try: 

160 fds.append(os.open(cwd, os.O_RDONLY | os.O_DIRECTORY)) 

161 fds.append(_open_dir(SAVE_DIR, fds[-1])) 

162 try: 

163 gi = os.open(".gitignore", os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 

164 0o600, dir_fd=fds[-1]) 

165 with os.fdopen(gi, "w") as fh: 

166 fh.write("*\n") 

167 except FileExistsError: 

168 pass 

169 fds.append(_open_dir(sub, fds[-1])) 

170 # A random suffix, retried on a clash: two turns saving the same file id 

171 # at once, or a temp left by a killed process, must not block a save. 

172 for _ in range(8): 

173 tmp = f".{name}.{os.urandom(6).hex()}.tmp" 

174 try: 

175 fd = os.open(tmp, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 

176 0o600, dir_fd=fds[-1]) 

177 break 

178 except FileExistsError: 

179 continue 

180 else: 

181 raise FileExistsError("no free temp file name") 

182 try: 

183 with os.fdopen(fd, "wb") as fh: 

184 fh.write(blob) 

185 os.replace(tmp, name, src_dir_fd=fds[-1], dst_dir_fd=fds[-1]) 

186 except BaseException: 

187 try: 

188 os.unlink(tmp, dir_fd=fds[-1]) 

189 except OSError: 

190 pass 

191 raise 

192 finally: 

193 for fd in fds: 

194 os.close(fd) 

195 retval = os.path.join(SAVE_DIR, sub, name) 

196 return retval

199def save_root(channel): 

200 """(cwd, None) where this conversation's attachments may be saved, or 

201 (None, reason) when they may not. 

202 

203 Only a conversation whose policy (directly, or through its `link`) sets a 

204 `cwd` of its own gets one: every other DM and channel shares 

205 default_policy's or exec.cwd's, 

206 so one person's file would be readable from another's turn. And never 

207 inside this deployment's own code, runtime files or standing prompt -- 

208 `exec.cwd` defaults to ".", which is the live checkout -- by the same 

209 lists the sandbox write-denies (#147, #174, self_roots).""" 

210 shared = (None, "this conversation has no working directory of its own, so it may " 

211 "be shared with others") 

212 try: 

213 # Its own `cwd`, not just an entry: a routing-only entry (mirror, link to 

214 # one), an empty one or one without `cwd` resolves to the shared default. 

215 resolved = policy.resolve(channel) 

216 cwd = policy.cwd_for(resolved) 

217 real = os.path.realpath(cwd) 

218 common = {os.path.realpath(policy.cwd_for(config.DEFAULT_POLICY)), 

219 os.path.realpath(os.path.expanduser(os.path.expandvars(config.EXEC_CWD)))} 

220 except Exception: # noqa: BLE001 -- fail closed: no save, said as a note 

221 logging.exception("attachments: could not resolve this conversation's working directory") 

222 return shared 

223 if not resolved.get("cwd") or real in common: 

224 return shared 

225 try: 

226 # Inside a self root (the checkout, venv, runtime files), or AT the 

227 # directory holding a config, policy or spine file. Only "at" for the 

228 # latter: a config file can sit in $HOME, and "under $HOME" would refuse 

229 # every channel on the host. 

230 inside = (any(sandbox._under(real, os.path.realpath(r)) for r in sandbox.self_roots()) 

231 or real in {os.path.dirname(os.path.realpath(p)) 

232 for p in tuple(config.SELF_FILES) + tuple(spine.files())}) 

233 except Exception: # noqa: BLE001 -- fail closed: no save, said as a note 

234 logging.exception("attachments: could not work out shmobster's own roots") 

235 inside = True 

236 if inside: 

237 retval = (None, "this channel's working directory is inside shmobster's own files") 

238 return retval 

239 retval = (cwd, None) 

240 return retval

243def _unsaved(name, why): 

244 """The model's line for a file it can read but has no saved copy of. Its 

245 own wording, not the could-not-read list: the content did arrive.""" 

246 retval = {"type": "text", "text": ( 

247 f"[attached file {name} was NOT saved to disk: {why}. There is no copy " 

248 f"to work from; do not retype it.]")} 

249 return retval

252def slack_source(f): 

253 """What to_parts needs to know about one Slack file: (name, mime, the 

254 per-file directory, a zero-argument fetch), or a str saying why it cannot 

255 be fetched. The fetch is Slack-only (_fetch).""" 

256 name = f.get("name") or "file" 

257 url = f.get("url_private_download") or f.get("url_private") 

258 if not url: 

259 retval = f"{name}: no download url in the event" 

260 return retval 

261 retval = (name, (f.get("mimetype") or "").lower(), _dir_id(f, url), lambda: _fetch(url)) 

262 return retval

265def to_parts(files, save_under=None, no_save=None, source=slack_source): 

266 """Turn one message's `files` into (content_parts, notes). 

267 

268 `notes` is every attachment we could not read and why -- surfaced to the 

269 user rather than swallowed, because silence here is exactly the bug that 

270 made #68 look like the model ignoring the question. 

271 

272 With `save_under` (the channel cwd), each file that passed the caps is 

273 also written to disk byte-identical and the model is told its path (#372): 

274 a model that only sees the text has no way to put the bytes in a file but 

275 to retype them, which silently dropped rows. A save that fails, or 

276 `no_save` (the reason this conversation gets none, from save_root), is said 

277 to the model per file, never dropped (#68). 

278 

279 `source` turns one entry of `files` into its name, type, directory and 

280 fetch (slack_source's contract). It is the only transport-specific part: 

281 the type and size caps, the content part and the save are the same for 

282 every ingest (#413). 

283 """ 

284 parts = [] 

285 notes = [] 

286 for f in files or []: 

287 got = source(f) 

288 if isinstance(got, str): 

289 notes.append(got) 

290 continue 

291 name, mime, sub, fetch = got 

292 if not (mime.startswith("image/") or mime.startswith("text/")): 

293 notes.append(f"{name}: unsupported type {mime or 'unknown'}") 

294 continue 

295 try: 

296 blob = fetch() 

297 except (urllib.error.URLError, ValueError, OSError) as exc: 

298 logging.warning("could not fetch attachment %s: %s", name, exc) 

299 notes.append(f"{name}: {exc}") 

300 continue 

301 if len(blob) > _MAX_BYTES: 

302 notes.append(f"{name}: larger than the {_MAX_BYTES} byte cap") 

303 continue 

304 parts.append(_part(name, mime, blob)) 

305 if no_save: 

306 parts.append(_unsaved(name, no_save)) 

307 elif save_under: 

308 try: 

309 path = _save(save_under, sub, _safe_name(name), blob) 

310 except OSError as exc: 

311 logging.warning("could not save attachment %s: %s", name, exc) 

312 parts.append(_unsaved(name, exc.strerror or str(exc))) 

313 continue 

314 parts.append({"type": "text", "text": ( 

315 f"[attached file {name} is saved byte-identical at {path}, relative " 

316 f"to the working directory ({len(blob)} bytes, sha256 " 

317 f"{hashlib.sha256(blob).hexdigest()}). Work from that path; copy it " 

318 f"before editing.]")}) 

319 retval = (parts, notes) 

320 return retval
