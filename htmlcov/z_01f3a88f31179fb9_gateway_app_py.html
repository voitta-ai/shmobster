1"""voitta-gateway ingest (#336): Signal, and whatever else the gateway carries, 

2as a second door into the same agent, beside Slack. 

3 

4The gateway owns the transport. It holds the one signal-cli process for the 

5shared account, decides which messages are addressed to this agent, and hands 

6over only those. This module polls them, runs each as a turn, and posts the 

7answer back as final text. The cross-cutting pieces a new ingest must wire 

8(CLAUDE.md, "Adding an ingest mode") are here too: 

9- approval cards, answered in text; 

10- the resume after a resolved request; 

11- skill proposals; 

12- the announcement, through slack_app's fan-out. 

13 

14Signal has no threads, so one conversation is one thread to approvals. 

15 

16Channel ids are `<medium>:<label>` for a group listed in the gateway's 

17conversations map, and `<medium>:id:<conversation_id>` for anything else (a 

18DM). User ids are `<medium>:<sender id>`, which is what `trusted_users` lists. 

19A sender id is an account identifier, so it is never put in a prompt or a 

20reply: identity.mention shows the name the gateway reported instead. 

21 

22Approvals are text, and they are not the model's call. A trusted user's whole 

23message reading `approve <id>` or `deny <id>` resolves that request here, the 

24way a button click does on Slack. Anything else is a turn, in which the model 

25can still call approve_command, under the same trusted-user check."""

93 mime = str(p.get("content_type") or "").lower() 

94 aid = p.get("attachment_id") 

95 name = p.get("filename") or (

135 busy = approvals.busy(channel, _THREAD) 

136 refs = [{k: f.get(k) for k in _FILE_REF_KEYS} for f in files or []]

219def send(channel, markdown, reply_to=None, key=None, raw=None, force=False): 

220 """Post a Markdown reply as final text, split to Signal's limit. Returns 

221 "ok", "retry" (may pass), "rejected" (the gateway refused it for good) or 

222 "withheld" (the group is not closed; see below). 

223 The first piece quotes `reply_to`, so a reply in a shared group stays this 

224 agent's: the gateway routes the answer to a quote back to whoever sent the 

225 quoted message. With `key`, each piece is idempotent, so posting the same 

226 thing again does not repeat the pieces that already went out. With `raw`, 

227 a (text, styles) pair is sent exactly as given: no Markdown is read. 

228 

229 Nothing reaches a Signal group that is not closed (CLAUDE.md), whatever the 

230 path -- a reply, command output after an approval, a card, an alert -- so 

231 the check is here, and a withheld post returns "withheld": not a gateway 

232 refusal, so nothing treats it as a reason to carry a task on. Only the 

233 refusal itself is `force`d.""" 

234 if not force and _group(channel) is not None and not _may_post(channel): 

235 return "withheld" 

236 retval = "ok" 

237 if raw: 

238 pieces = gateway_text.raw_parts(*raw) 

239 else: 

240 plain = unsigned(for_signal(markdown)).replace(_MARK, "") # after the marker is gone 

241 text, ids = _to_mentions(channel, plain) 

242 pieces = gateway_text.parts(text) 

243 marks = sum(p.get("text", "").count(_MARK) for parts in pieces for p in parts) 

244 if marks != len(ids): # rendering moved or dropped one: no mention beats a wrong one 

245 pieces, ids = gateway_text.parts(plain), [] 

246 pieces = _attach_mentions(pieces, ids) 

247 for i, parts in enumerate(pieces): 

248 body = {"to": _target(channel), "parts": parts} 

249 if reply_to and i == 0: 

250 body["reply_to"] = reply_to 

251 if key: 

252 body["idempotency_key"] = f"{key}:{i}" 

253 try: 

254 _call("POST", "/v1/messages", body) 

255 except urllib.error.HTTPError as exc: 

256 logging.warning("gateway: send to %s refused: %s %s", channel, exc.code, _error_code(exc)) 

257 if exc.code < 500 and exc.code != 429: 

258 return "rejected" 

259 retval = "retry" 

260 except Exception as exc: # noqa: BLE001 

261 logging.warning("gateway: send to %s failed: %s", channel, type(exc).__name__) 

262 retval = "retry" 

263 return retval

269def _card_command(cmd): 

270 """The command as the card shows it, always within one Signal message. Split 

271 across messages, the cut points would hide whether a newline or a space stood 

272 there, and two different commands could show the same card (#386 review). Too 

273 long for one message: its head and tail exactly as written, and in between a 

274 line no command contains, saying how much is not shown.""" 

275 if gateway_text.size(cmd) <= _CARD_BUDGET: 

276 return cmd 

277 half = _CARD_BUDGET // 2 - 100 

278 head, tail = _bytes_prefix(cmd, half), _bytes_suffix(cmd, half) 

279 hidden = gateway_text.size(cmd) - gateway_text.size(head) - gateway_text.size(tail) 

280 retval = (f"{head}\n\u2702\ufe0f [{hidden} bytes of this command are not shown here: read the whole " 

281 f"command on the host before approving]\n{tail}") 

282 return retval

381def members(channel, fresh=False): 

382 """The gateway's report on a group: members (id, admin, self, person_id, 

383 agent) and whether it is open. Cached a minute (a failure a few seconds); 

384 None when it could not be had or is not the expected shape, which serves 

385 nothing (fail closed).""" 

386 label = _group(channel) 

387 if label is None: 

388 return None 

389 now = time.time() 

390 with _LOCK: 

391 hit = _MEMBERS.get(channel) 

392 if hit and not fresh and now - hit[0] < (_MEMBERS_TTL if hit[1] is not None else _MEMBERS_FAIL_TTL): 

393 return hit[1] 

394 try: 

395 info = _call("GET", f"/v1/conversations/{urllib.parse.quote(label, safe='')}/members", timeout=10) 

396 if not (isinstance(info, dict) and isinstance(info.get("members"), list) 

397 and isinstance(info.get("open"), bool) and isinstance(info.get("reasons", []), list)): 

398 info = None 

399 except urllib.error.HTTPError as exc: 

400 logging.warning("gateway: could not list members of %s: %s %s", channel, exc.code, _error_code(exc)) 

401 info = None 

402 except Exception as exc: # noqa: BLE001 

403 logging.warning("gateway: could not list members of %s (%s)", channel, type(exc).__name__) 

404 info = None 

405 with _LOCK: 

406 _MEMBERS[channel] = (now, info) 

407 return info

438def _may_post(channel): 

439 """Whether a post may go to this group now. A failed read is tried once more 

440 before the post is withheld, so a blip does not cost a closed group its answer.""" 

441 why = _closed(channel) 

442 if why and members(channel) is None: 

443 time.sleep(1) 

444 members(channel, fresh=True) 

445 why = _closed(channel)

447 logging.warning("gateway: withheld a post to %s: %s", channel, "; ".join(why)) 

448 retval = not why 

449 return retval

452def _guard(fn, data, *args): 

453 """Run `fn(*args)` for a message, unless its group is open (refused, said at 

454 most hourly) or it comes from another agent that has had its run of answers 

455 since a person last spoke here (dropped, logged).""" 

456 channel = channel_of(data) 

457 why = _closed(channel) 

458 if why is not None: 

459 logging.warning("gateway: not serving %s: %s", channel, "; ".join(why)) 

460 now = time.time() 

461 with _LOCK: 

462 due = now - _REFUSED.get(channel, 0) >= _REFUSE_EVERY 

463 if due: 

464 _REFUSED[channel] = now 

465 if due: 

466 send(channel, "I only work in closed groups, and I can't serve this one: " 

467 + "; ".join(why) + ". A group admin can turn the group link off and set " 

468 "Add members to Only admins; then ask me again.", 

469 reply_to=data.get("message_id"), force=True) 

470 return 

471 if _group(channel) is not None: 

472 sender_id = (data.get("sender") or {}).get("id") 

473 sender = _member(channel, sender_id) 

474 if sender is None: # joined since the cache was filled: ask again 

475 info = members(channel, fresh=True) 

476 sender = next((m for m in (info or {}).get("members") or [] if m.get("id") == sender_id), None) 

477 with _LOCK: 

478 if sender and sender.get("agent"): 

479 run = _AGENT_RUN.get(channel, 0) 

480 if run >= _AGENT_RUN_MAX: 

481 logging.warning("gateway: not answering another agent in %s: %d answers in a row " 

482 "since a person spoke", channel, run) 

483 return 

484 _AGENT_RUN[channel] = run + 1 

485 else: 

486 _AGENT_RUN[channel] = 0 

487 fn(*args)

490def _to_mentions(channel, text): 

491 """`@<name>` in a reply, for a member of this group, becomes a real Signal 

492 mention: the U+FFFC Signal expects, plus the member ids in order.""" 

493 info = members(channel) if _group(channel) else None 

494 names = {} 

495 for m in (info or {}).get("members") or []: 

496 if m.get("id") and not m.get("self"): 

497 names[_name(channel, m["id"])] = m["id"] 

498 ids = [] 

499 text = text.replace(_MARK, "") # only the mentions made here may carry one 

500 if names: 

501 alt = "|".join(re.escape(n) for n in sorted(names, key=len, reverse=True)) 

502 pattern = re.compile(rf"(?<![\w/.@:])@({alt})(?![\w-])") 

503 def one(match): 

504 ids.append(names[match.group(1)]) 

505 return _MARK 

506 # Not inside code (``` blocks, `spans`): a name there is text, not a mention. 

507 chunks = re.split(r"(```.*?```|`[^`\n]*`)", text, flags=re.S) 

508 text = "".join(c if i % 2 else pattern.sub(one, c) for i, c in enumerate(chunks)) 

509 return text, ids

512def _attach_mentions(pieces, ids): 

513 """Each U+FFFC left in the rendered parts, in order, gets the next id.""" 

514 ids = list(ids) 

515 for parts in pieces: 

516 for part in parts: 

517 if part.get("type") != "text" or _MARK not in part.get("text", ""): 

518 continue 

519 at, found = 0, [] 

520 for ch in part["text"]: 

521 if ch == _MARK and ids: 

522 found.append({"start": at, "length": 1, "id": ids.pop(0)}) 

523 at += len(ch.encode("utf-16-le")) // 2 

524 if found: 

525 part["mentions"] = found 

526 return pieces

529def _mentioned(text, mentions, channel=None): 

530 """Signal puts each @mention on one U+FFFC. A mention of this account reads as 

531 @<agent label> -- the account's profile name is shared by every agent on it, and 

532 the model knows itself by its label -- and anyone else by the name _name gives 

533 them in this conversation (@someone when there is no conversation to ask).""" 

534 units = text.encode("utf-16-le") 

535 valid = [m for m in mentions if isinstance(m, dict) and type(m.get("start")) is int and m["start"] >= 0] 

536 for m in sorted(valid, key=lambda m: m["start"], reverse=True): 

537 at = m["start"] 

538 if units[2 * at:2 * at + 2] != _MARK.encode("utf-16-le"): 

539 continue 

540 if m.get("self"): 

541 name = f"@{config.AGENT_LABEL or 'me'}" 

542 else: 

543 name = f"@{_name(channel, m.get('id'))}" if channel and m.get("id") else "@someone" 

544 units = units[:2 * at] + name.encode("utf-16-le") + units[2 * at + 2:] 

545 retval = units.decode("utf-16-le").replace(_MARK, "@someone") 

546 return retval

549def _text(data, channel=None, names=True, files=True): 

550 """The message as text: text parts, and voice notes by their transcript. 

551 `channel` names other people's mentions; an event carries its own. With 

552 `names` off (the poll thread's parsing) nobody is looked up. With `files` 

553 off, other file parts are left to the caller, which reads them (#413); 

554 otherwise they are named as unread, which is right for history.""" 

555 if channel is None and names and data.get("conversation"): 

556 channel = channel_of(data) 

557 pieces, unread = [], [] 

558 mentions = data.get("mentions") or [] 

559 for p in data.get("parts") or []: 

560 if p.get("type") == "text": 

561 pieces.append(_mentioned(p.get("text", ""), mentions, channel)) 

562 mentions = [] # offsets are into the message's one text part 

563 elif p.get("transcript"): 

564 pieces.append(f"[voice note] {p['transcript']}") 

565 elif p.get("type") == "file" and files: 

566 unread.append(p.get("filename") or p.get("content_type") or "a file") 

567 retval = "\n".join(pieces).strip() 

568 if unread: 

569 retval += "\n\n[attachments I could not read: " + "; ".join(unread) + "]" 

570 return retval

573def _history(conversation_id, current, medium="signal", channel=None): 

574 """The conversation so far, labelled by speaker, as slack_app does for a 

575 thread. Best-effort: no history is a turn without context, not a failure.""" 

576 try: 

577 q = urllib.parse.urlencode({"limit": _HISTORY}) 

578 msgs = _call("GET", f"/v1/conversations/{conversation_id}/history?{q}").get("messages", []) 

579 except Exception: # noqa: BLE001 

580 return None 

581 lines = [] 

582 for m in msgs: 

583 if m.get("message_id") == current: 

584 continue 

585 sender = m.get("sender") or {} 

586 text = _text(m, channel) 

587 if sender.get("kind") == "self": 

588 consumer = sender.get("consumer") 

589 mine = consumer == config.GATEWAY_CONSUMER 

590 if mine: 

591 who, text = f"{config.AGENT_LABEL or 'me'} (me)", unsigned(text) 

592 elif consumer: 

593 who = f"{consumer} (another agent)" 

594 else: # the gateway's own reply, e.g. to /help 

595 who = "the gateway (this account's notice, not an agent)" 

596 if mine and _MIRRORED in text: 

597 who = "mirrored from Slack, not my words" 

598 else: 

599 # By the operator's name or a handle, and trust -- never the profile 

600 # name the sender chose (#336 review) 

601 user = f"{medium}:{sender.get('id')}" 

602 who = (_name(channel, sender.get("id")) if channel else identity.handle(user)) 

603 who += " (another agent)" if channel and (_member(channel, sender.get("id")) or {}).get("agent") else "" 

604 who += " (trusted)" if admin_tools.is_trusted(user) else "" 

605 if text: 

606 lines.append(f"[{who}] {text}") 

607 retval = "\n".join(lines) or None 

608 return retval

611def _unaddressed(text): 

612 """The text without how it was addressed: Signal's U+FFFC for an 

613 @mention, or the agent's name, so "@Ada approve <id>" is an approval.""" 

614 names = [n for n in (config.AGENT_LABEL,) if n] 

615 lead = r"^[\s\ufffc,:]*(?:@?(?:" + "|".join(re.escape(n) for n in names) + r")\b[\s,:]*)*" if names else r"^[\s\ufffc,:]*" 

616 retval = re.sub(lead, "", text, flags=re.IGNORECASE) 

617 return retval

696 except Exception: # noqa: BLE001 

697 pass

701 user_id=user, surface=surface, attachments=parts) 

702 except Exception as exc: # one short line; the exception goes to the log only (#352) 

703 logging.exception("gateway: handler failed: channel=%s message=%s", channel, data["message_id"])

706 _post_pending(channel, surface) 

707 mirror_out(channel, identity.handle(user), text, reply)

710def _resolve_text(approved, req_id, channel, user, surface): 

711 """`approve <id>` / `deny <id>` from a trusted user: what a click is on Slack.""" 

712 req_id = approvals.canonical(req_id) 

713 req = approvals.acquire(req_id, channel) 

714 if req is None: 

715 if approvals.status(req_id, channel)[0] != "held": # held: another surface reports it 

716 send(channel, f"[{req_id}] is not pending here: it was already acted on, it is from " 

717 "before a restart, or it belongs to another conversation.") 

718 return 

719 ctx = {"user_id": user, "channel": channel, "thread_ts": _THREAD, "surface": surface} 

720 try: 

721 run = admin_tools.run_approved if approved else admin_tools.run_denied 

722 result = redact.scrub(run(req_id, req, ctx)) 

723 finally: 

724 approvals.release(req_id) 

725 send(channel, None, raw=_verbatim("", result[:_CARD_COMMAND_MAX * 2])) 

726 _resume(channel, req_id, req, approved, result, user, surface)

729def _resume(channel, req_id, req, approved, result, user, surface): 

730 """Carry the turn on after a request resolved (#169); handler.resume decides 

731 whether this is the moment.""" 

732 try: 

733 cid = _CONVERSATIONS.get(channel) 

734 # The task's files, read again only when this answer can resume it: 

735 # with other requests still parked, handler.resume refuses. 

736 parts, notes = ([], []) if approvals.pending_in(channel, _THREAD) \ 

737 else _read_files(channel, _task_files(channel)) 

738 reply = handler.resume(req_id, approved, req.get("command", ""), result, 

739 attachments=parts, attachment_notes=notes, 

740 thread_context=_history(cid, None, channel=channel) if cid else None, 

741 channel=channel, thread_ts=_THREAD,

743 if reply is None: 

744 waiting = approvals.pending_in(channel, _THREAD) 

745 if waiting: 

746 send(channel, f"[{req_id}] is resolved, and this task is waiting on {waiting} more " 

747 f"request{'s' if waiting > 1 else ''} here. It continues on its own " 

748 "once the last one is answered.") 

749 return 

750 send(channel, reply) 

751 _post_pending(channel, surface) 

752 except Exception: 

753 logging.exception("gateway: could not resume after [%s]", req_id)

770def _post_pending(channel, surface): 

771 """Cards for what the turn parked, as text a trusted user answers in text.""" 

772 for req_id, req in approvals.claim_unsurfaced(channel, _THREAD): 

773 cmd = _card_command(redact.scrub(req["command"])) 

774 reason = f" ({redact.scrub(req['reason'])})" if req.get("reason") else "" 

775 # A refusal and a question must not look alike (#172) 

776 head = (f"\u26d4 REFUSED by the classifier -- approve only if you are sure [{req_id}]{reason}" 

777 if req.get("refused") else f"\U0001F512 Needs approval [{req_id}]{reason}") 

778 card = _verbatim(head, cmd, _how_to_answer(channel, req_id)) 

779 if gateway_text.size(card[0]) > gateway_text.LIMIT: # a reason long enough to push it over: 

780 short = (f"\u26d4 REFUSED by the classifier -- approve only if you are sure [{req_id}]" 

781 if req.get("refused") else f"\U0001F512 Needs approval [{req_id}]") 

782 card = _verbatim(short, cmd, _how_to_answer(channel, req_id)) # one message 

783 if gateway_text.size(card[0]) > gateway_text.LIMIT: # a card is never split (#386 review) 

784 _card_failed(channel, req_id, surface, rejected=True) 

785 continue 

786 outcome = send(channel, None, key=f"card:{req_id}", raw=card) 

787 if outcome == "withheld": # the group is not closed: drop the request, carry nothing on 

788 approvals.withdraw(req_id) 

789 logging.warning("gateway: withdrew [%s]: %s is not closed", req_id, channel) 

790 elif outcome != "ok": 

791 _card_failed(channel, req_id, surface, rejected=outcome == "rejected") 

792 for key, prop in proposals.claim_unsurfaced(channel, _THREAD): 

793 card = redact.scrub(f"Worth a skill? [{key}] {prop['name']} -- {prop['why']}\n" 

794 f"A trusted user can ask me to open it ({key}) or decline it.") 

795 if send(channel, card, key=f"proposal:{key}") != "ok": 

796 proposals.unsurface(key)

799def _card_failed(channel, req_id, surface, rejected=False): 

800 """A card did not post: retry on the next turn when the failure may pass, 

801 up to _CARD_TRIES; withdraw it when the gateway refused it or the retries 

802 ran out, as slack_app does (#356), so it cannot hold the thread.""" 

803 if not rejected and approvals.unsurface(req_id) < _CARD_TRIES: 

804 return 

805 if approvals.withdraw(req_id) is None: 

806 return 

807 send(channel, f"[{req_id}] needed approval, but its card could not be posted, so it was " 

808 "withdrawn and did not run. Ask again if it is still needed.") 

809 outcome = approvals.take_deferred(channel, _THREAD) 

810 if outcome is not None: 

811 _resume(channel, outcome["req_id"], {"command": outcome["command"], "requester": outcome["requester"]}, 

812 outcome["approved"], outcome["result"], outcome["user_id"], surface)

822def _begin(mid, started): 

823 """Record that this message's turn is starting, before anything runs. A turn 

824 that began is never run again after a restart -- not even one that ended the 

825 process -- while one still waiting in the queue is (#386 review).""" 

826 with _STARTED_LOCK: # append and save together: no older copy saved last 

827 started.append(mid) 

828 del started[:-_STARTED_MAX]

837def _loop(): 

838 """Poll, hand each message to a worker, and ack (persist the cursor) only 

839 up to the oldest message whose turn has not finished, so a restart re-reads 

840 what was still queued. Turns that had started are skipped on that re-read 

841 (`_STARTED`): at most once for anything that may have acted, at least once 

842 for the rest.""" 

843 acked = fetched = state.get(_CURSOR) or "now" 

844 started = list(state.get(_STARTED) or []) 

845 inflight = [] # [(event id, future or None)], in order 

846 while True: 

847 try: 

848 while inflight and (inflight[0][1] is None or inflight[0][1].done()): 

849 acked = inflight.pop(0)[0] 

850 if not inflight and fetched != acked: 

851 acked = fetched 

852 if acked != state.get(_CURSOR): 

853 state.put(_CURSOR, acked) # persisting the cursor is the ack (gateway README) 

854 q = urllib.parse.urlencode({"after": fetched, "wait": 25, "types": "message.received"}) 

855 body = _call("GET", f"/v1/events/poll?{q}", timeout=40) 

856 if not isinstance(body, dict): 

857 raise ValueError("poll answered with something other than an object") 

858 for e in body.get("events") or []: 

859 if not isinstance(e, dict) or not isinstance(e.get("data"), dict): 

860 continue 

861 data, fut = e["data"], None 

862 mid = data.get("message_id") 

863 if (e.get("type") == "ai.voitta.message.received.v1" and isinstance(mid, str) 

864 and mid not in started and _first_time(mid)): 

865 approval = _approval(data) 

866 # Every lane goes through _guard: no open group is served (CLAUDE.md). 

867 if approval: # its own lane: never behind long turns 

868 fut = _approvals_pool.submit(_run, _guard, mid, started, _resolve_text, data, *approval) 

869 elif (help_in := _help(data)): # no model: the same lane, it is as quick 

870 fut = _approvals_pool.submit(_run, _guard, mid, started, _describe, data, help_in, mid) 

871 else: 

872 fut = _pool.submit(_run, _guard, mid, started, _turn, data, data) 

873 if isinstance(e.get("id"), str): 

874 inflight.append((e["id"], fut)) 

875 if isinstance(body.get("next"), str): 

876 fetched = body["next"] 

877 except Exception as exc: # noqa: BLE001 -- this thread must not die (#66: deaf but alive) 

878 logging.warning("gateway: poll failed (%s); retrying", type(exc).__name__) 

879 time.sleep(5)

882def start(slack_post=None): 

883 """Start polling the gateway in the background, when one is configured.""" 

884 global _slack_post 

885 if not config.GATEWAY_URL: 

886 return False

889 served = conversations(timeout=5) # boot waits on this; a hung gateway must not stall it 

890 if served is not None:

899 if ":" not in key or ":id:" in key:

902 logging.warning("gateway: policy %s names a conversation the gateway does not route to " 

903 "consumer %r (gateway.consumer must match the token's)", key, config.GATEWAY_CONSUMER) 

904 ignored = sorted(set(pol) - set(policy.ROUTING_KEYS)) if pol.get("link") else [] 

905 if ignored: 

906 logging.warning("gateway: policy %s is linked to %s, so its %s are ignored", 

907 key, pol["link"], ", ".join(ignored))
