1"""Git over https for every channel, so nothing ever needs to read ~/.ssh. 

2 

3ssh needs three files under ~/.ssh: config, known_hosts (host verification is 

4mandatory without a terminal) and a private key, or an agent that holds one. 

5Granting the directory to a channel grants a read-only `cat ~/.ssh/id_*` that 

6runs without a card; denying it breaks `git push`. Neither is acceptable, so 

7the channel's git does not use ssh at all. 

8 

9Git >= 2.31 takes configuration from the environment (GIT_CONFIG_COUNT and 

10GIT_CONFIG_KEY_n / GIT_CONFIG_VALUE_n), scoped to that one process: the 

11operator's repos and ~/.gitconfig are untouched, and the entries win over 

12the global and system config. Four entries: 

13 

14- url.https://github.com/.insteadOf = git@github.com: and ssh://git@github.com/ 

15 Every remote that names github over ssh is used over https. This also 

16 overrides the opposite rewrite an operator commonly carries in 

17 ~/.gitconfig (url.ssh://git@github.com/.insteadOf https://github.com/). 

18- credential.helper = "" then !gh auth git-credential 

19 The empty value resets the helper list; without it git also runs any 

20 helper configured system-wide -- Homebrew ships credential.helper 

21 osxkeychain, which hangs inside the sandbox waiting on a keychain prompt. 

22 gh keeps its token in the keychain, which the sandbox does not restrict, 

23 and gh itself already runs in every channel. 

24 

25GIT_TERMINAL_PROMPT=0 goes with them: a git that would otherwise ask for a 

26password now fails at once instead of holding the command until the timeout. 

27 

28Verified under the sandbox with ~/.ssh fully denied: `git ls-remote` and an 

29authenticated `git push --dry-run` both succeed.""" 

30import os 

31import subprocess 

32 

33_ENTRIES = ( 

34 ("url.https://github.com/.insteadOf", "git@github.com:"), 

35 ("url.https://github.com/.insteadOf", "ssh://git@github.com/"), 

36 ("credential.helper", ""), 

37 ("credential.helper", "!gh auth git-credential"), 

38)

41def env(): 

42 """The environment additions, as a dict to merge over a channel's env.""" 

43 retval = {"GIT_TERMINAL_PROMPT": "0", "GIT_CONFIG_COUNT": str(len(_ENTRIES))} 

44 for i, (key, value) in enumerate(_ENTRIES): 

45 retval[f"GIT_CONFIG_KEY_{i}"] = key 

46 retval[f"GIT_CONFIG_VALUE_{i}"] = value 

47 return retval

50def _run(argv, extra_env=None): 

51 """(returncode, stdout) or (None, "") when the binary is missing or hangs.""" 

52 retval = (None, "") 

53 try: 

54 proc = subprocess.run( 

55 argv, capture_output=True, text=True, timeout=10, 

56 env={**os.environ, **(extra_env or {})}, 

57 ) 

58 retval = (proc.returncode, proc.stdout) 

59 except (OSError, subprocess.SubprocessError): 

60 retval = (None, "") 

61 return retval

64def preflight(): 

65 """Warnings for a host that cannot run git the way env() assumes: a git 

66 older than 2.31 ignores GIT_CONFIG_COUNT and would quietly go back to ssh 

67 and the system credential helpers; a gh that is not logged in cannot 

68 answer git's credential request. Startup logs these once.""" 

69 retval = [] 

70 rc, out = _run(["git", "config", "--get", "shmobster.preflight"], {**env(), "GIT_CONFIG_COUNT": "1", 

71 "GIT_CONFIG_KEY_0": "shmobster.preflight", 

72 "GIT_CONFIG_VALUE_0": "ok"}) 

73 if rc is None: 

74 retval.append("git is not runnable") 

75 elif out.strip() != "ok": 

76 retval.append("git ignores GIT_CONFIG_COUNT (needs git >= 2.31); channels would fall back to ssh") 

77 rc, _ = _run(["gh", "auth", "status", "--hostname", "github.com"]) 

78 if rc is None: 

79 retval.append("gh is not runnable; git over https has no credential helper") 

80 elif rc != 0: 

81 retval.append("gh is not logged in to github.com; git push will fail in every channel") 

82 return retval
