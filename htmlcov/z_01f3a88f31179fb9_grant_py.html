1"""Grant layer (#117): run in-tree writes and self-authored commits without a card. 

2 

3YOLT answers "does this mutate anything?" and every yes used to park for a 

4human -- including `cp x app/index.html && git add ... && git commit ...` 

5inside the channel's own worktree, the work the channel exists for. That 

6relief cannot come from YOLT: it is being narrowed to deny-only 

7(voitta-yolt#98) and its reasons are about Claude Code as host. shmobster is 

8an executor with its own approval flow and no classifier in front of it 

9(voitta-yolt#114, tier 2), so the grant lives here, next to the only party 

10that knows the channel's tree. 

11 

12A YOLT-unsafe command is granted when EVERY segment is one of: 

13 

14- YOLT-safe on its own (`cd`, `diff`, `git status`, `git log`, ...); 

15- a filesystem verb (cp, mv, mkdir, touch, tee, ln, chmod, sed) -- granted 

16 on the verb, not the operands, so the boundary is exactly the sandbox's 

17 write roots (#116): the tree, its worktrees sibling, the temp dir and the 

18 toolchain caches. A write to /tmp runs without a card; a write anywhere 

19 else fails in the kernel. Without the sandbox this module must not grant; 

20- a local git write: `git add`, `git mv`, `git stash`, `git checkout -b`, 

21 `git switch -c`, and `git commit` when the tracked directory is a linked 

22 worktree on a non-default branch whose commits are all the user's own 

23 (gitstate.py, #82's predicates); 

24- one of the above with an output redirect (`cat > f <<EOF` is how an agent 

25 writes a file), as long as the target is not a device other than 

26 /dev/null, /dev/stdout, /dev/stderr. 

27 

28Everything else parks as before. This is an allowlist, so the tree-local 

29destructive verbs are excluded by construction: rm, git reset, git clean, 

30git checkout -- <path>, git restore, git push, git branch -D, gh, aws, sudo, 

31env, bash -c, python. Committed work is reflog-recoverable; uncommitted work 

32was the operator's call, and the operator made it (#117): card. 

33 

34The command is walked as a bash AST (tree-sitter, already a dependency via 

35YOLT), not tokenized: quoting, heredocs and continuation lines come out 

36right, and a node kind this module does not model -- subshell, variable 

37assignment, if/while, function -- is refused rather than guessed at. A plain 

38`for` loop and a `--`-guarded command substitution are modelled, as reads only 

39(#373). A literal `cd` moves the tracked directory in 

40source order, starting from the channel cwd; a `cd` that cannot be resolved 

41statically (`cd "$DIR"`, `cd -`) makes every later `git commit` ungrantable 

42rather than judged against the wrong tree. `git -C <path>` retargets one 

43segment the same way."""

226def _file_refusal(texts): 

227 """`file -C` compiles the magic files it is given into `<name>.mgc` beside 

228 them -- a write, so `file` cannot be judged on its verb alone (#373 

229 review). getopt_long takes any unambiguous prefix, so `--comp` is 

230 `--compile` too, and a cluster such as `-Cm` carries it. 

231 

232 Every word is scanned, `--` or not: in `file -m -- -C` the `--` is the 

233 value of `-m` and `-C` is still an option, and telling the two apart means 

234 knowing every flag that takes a value. Scanning all of argv only parks a 

235 file literally named like the flag.""" 

236 for t in texts: 

237 if t.startswith("--"): 

238 name = t.split("=", 1)[0] 

239 if len(name) > 2 and "--compile".startswith(name): 

240 return f"file: {name} writes a compiled magic file" 

241 elif t.startswith("-") and "C" in t[1:]: 

242 return f"file: {t} writes a compiled magic file" 

243 return None

334def _awk_program_refusal(prog): 

335 """Why this awk program is not one of the read-only ones, or None.""" 

336 i, n = 0, len(prog) 

337 prev = "" 

338 while i < n: 

339 c = prog[i] 

340 if c == '"': 

341 j = i + 1 

342 while j < n and prog[j] != '"': 

343 j += 2 if prog[j] == "\\" else 1 

344 if j >= n: 

345 return "awk: unterminated string" 

346 i, prev = j + 1, "x" 

347 continue 

348 if c == "/" and prev not in ("x", ")", "]"): 

349 # A regex literal: here `/` cannot be division, so it opens one. 

350 j = i + 1 

351 while j < n and prog[j] != "/": 

352 j += 2 if prog[j] == "\\" else 1 

353 if j >= n: 

354 return "awk: unterminated regex" 

355 i, prev = j + 1, "x" 

356 continue 

357 if c.isalpha() or c == "_": 

358 j = i 

359 while j < n and (prog[j].isalnum() or prog[j] == "_"): 

360 j += 1 

361 word = prog[i:j] 

362 k = j 

363 while k < n and prog[k] in " \t": 

364 k += 1 

365 if word in _AWK_REFUSED_NAMES: 

366 return f"awk: {word}" 

367 if k < n and prog[k] == "(" and word not in _AWK_CALLS and word not in ("if", "while", "for"): 

368 return f"awk: calls {word}()" 

369 if word in ("print", "printf"): 

370 prev = "print" 

371 else: 

372 prev = "x" 

373 i = j 

374 continue 

375 if c in "+-" and prog[i:i + 2] in ("++", "--"): 

376 # Postfix after an operand keeps it an operand, so a `/` next is 

377 # division; prefix leaves an operator before the operand to come. 

378 i += 2 

379 prev = "x" if prev == "x" else "op" 

380 continue 

381 if c == "|": 

382 if prog[i:i + 2] != "||": 

383 return "awk: pipe" 

384 i += 2 

385 prev = "op" 

386 continue 

387 if c not in _AWK_CHARS: 

388 return f"awk: {c!r} outside a string or regex" 

389 if c not in " \t\n\r": 

390 prev = "x" if c.isdigit() or c in ")]$" else ("op" if c != "." else prev) 

391 i += 1 

392 return None

419# `date` prints a time unless it is asked to set one, and what counts as asking 

420# differs by implementation. BSD getopt stops at the first non-option and BSD 

421# then treats a following word as the time to set, where GNU permutes; a `+` 

422# word is the format only in BSD's first position; and without `-j`, BSD's `-f` 

423# exists only to parse a time to set. So this reads the strictest common shape: 

424# allowlisted options first, nothing dash-shaped after them, and then either 

425# `-j` among those options (parse, never set) or at most one `+format`. The 

426# conversions an agent reaches for -- `date -u -r 1770068151`, `date -d @1770068151`, 

427# `date -j -f %s 1770068151 +%F` -- used to card.

429_DATE_VALUE_FLAGS = frozenset("rdvfz") 

430_DATE_LONG = frozenset(( 

431 "--utc", "--universal", "--rfc-email", "--rfc-2822", "--iso-8601", 

432 "--rfc-3339", "--debug", "--help", "--version", "--date", "--reference", 

433 "--file", 

434)) 

435_DATE_LONG_VALUE = frozenset(("--date", "--reference", "--file"))

438def _date_refusal(texts): 

439 """`date` whose words read the same to BSD and GNU, and print on both.""" 

440 no_set = False 

441 parses_input = False 

442 i = 0 

443 while i < len(texts) and texts[i].startswith("-") and len(texts[i]) > 1: 

444 t = texts[i] 

445 i += 1 

446 if t.startswith("--"): 

447 if t.split("=")[0] not in _DATE_LONG: 

448 return f"date: option {t}" 

449 if t in _DATE_LONG_VALUE: 

450 i += 1 

451 continue 

452 for k, ch in enumerate(t[1:], 1): 

453 if ch == "j": 

454 no_set = True 

455 if ch in _DATE_FLAGS: 

456 continue 

457 if ch == "I": 

458 break # -I[FMT]: the rest of the word is its argument 

459 if ch in _DATE_VALUE_FLAGS: 

460 parses_input = parses_input or ch == "f" 

461 if k == len(t) - 1: 

462 i += 1 # its value is the next word 

463 break 

464 return f"date: option -{ch}" 

465 rest = texts[i:] 

466 if any(t.startswith("-") and len(t) > 1 for t in rest): 

467 return "date: an option after a positional reads differently on BSD and GNU" 

468 if no_set: 

469 return None 

470 if parses_input: 

471 return "date: -f without -j sets the clock" 

472 setting = rest[1:] if rest[:1] and rest[0].startswith("+") else rest 

473 if setting: 

474 return f"date: {setting[0]} would set the clock" 

475 return None

538def _quoted_non_flag(raw): 

539 """True when this argument cannot be an option, although it expands. 

540

555def _fetch_write_flag(texts): 

556 """The flag making this fetch write a file, or None. 

557 

558 Clusters count: `-sO url` saves to a file as surely as `-O url`, and this 

559 is the same reasoning `_sort_refusal` spells out -- a single-dash cluster 

560 is a set of option letters, not a word. 

561 

562 The three devices are not files, the same exception the redirect rule makes 

563 (`_DEV_OK`): `curl -o /dev/null -w '%{http_code}'` asks for a status code 

564 and keeps the body, which is the cheapest read there is and was costing a 

565 card. The destination is only trusted when it is the next word, attached 

566 (`-o/dev/null`) or after `=`; a cluster like `-sO` names no destination at 

567 all, so it stays refused.""" 

568 retval = None 

569 for i, t in enumerate(texts): 

570 head = t.split("=", 1)[0] 

571 if head in _FETCH_WRITE_LONG or (t.startswith("-") and not t.startswith("--") 

572 and set(t[1:]) & _FETCH_WRITE_LETTERS): 

573 if _writes_device_only(t, texts[i + 1:], head): 

574 continue 

575 retval = head if head in _FETCH_WRITE_LONG else t 

576 break 

577 return retval

594def _writes_device_only(token, after, head): 

595 """True when this output flag's destination is /dev/null or a standard 

596 stream, so nothing is written to a file after all. 

597 

598 Only `-o` / `--output` name a destination this can read. `-O` takes the 

599 name from the URL and `-J` from a response header, so a token carrying 

600 either is never exempt however it is spelled. 

601 

602 The short form is read the way curl reads it: in a cluster, `o` consumes 

603 the rest of the token if there is any (`-so/dev/null`, `-o/dev/null`) and 

604 otherwise the next word (`-so /dev/null`).""" 

605 if head == "--output" or token.startswith("--output="): 

606 dest = token.split("=", 1)[1] if "=" in token else (after[0] if after else None) 

607 elif token.startswith("-") and not token.startswith("--"): 

608 cluster = token[1:] 

609 if set(cluster) & set("OJ") or "o" not in cluster: 

610 return False 

611 tail = cluster.split("o", 1)[1] 

612 dest = tail if tail else (after[0] if after else None) 

613 else: 

614 return False 

615 retval = dest in _DEV_OK 

616 return retval

908def _fd_dup(node): 

909 """True when a file_redirect only moves or closes a descriptor. 

910 

911 The distinction is bash's and it is genuinely ambiguous in the text: 

912 `>&word` duplicates a descriptor when `word` is a number, and redirects 

913 both streams into a *file* named `word` when it is not. So `>&2` writes 

914 nothing and `>&out.txt` writes a file, spelled with the same operator. 

915 

916 We do not re-implement that rule -- tree-sitter has already applied it, and 

917 types the two destinations `number` and `word`. Reading the node type is 

918 therefore asking the parser what bash decided, rather than asking a regex 

919 what the string looks like. Anything that is neither (`$X`, unparsed, a 

920 destination we do not recognize) falls through to the write path, which is 

921 the fail-closed direction and where `&>file` stays too.""" 

922 types = [c.type for c in node.children] 

923 if any(t in _CLOSE_OPS for t in types): 

924 retval = True 

925 else: 

926 retval = (any(t in _DUP_OPS for t in types) 

927 and bool(node.children) 

928 and node.children[-1].type == "number") 

929 return retval

1018def _literal_word(node, src): 

1019 """The word bash ends up with, when that takes no shell rules beyond quote 

1020 removal to work out, else None. A check has to read what the command gets 

1021 rather than what the source looks like, so quoted pieces are joined, a 

1022 double-quoted string's escapes are decoded (_dq_unescape), and an unquoted 

1023 word with a glob, a brace or an escape is not read at all. The escaped 

1024 parentheses and `!` of a `find` expression are the one exception.""" 

1025 text = _text(node, src) 

1026 retval = None 

1027 if node.type == "raw_string": 

1028 retval = text[1:-1] 

1029 elif node.type in ("word", "number"): 

1030 if text in ("\\(", "\\)", "\\!"): 

1031 retval = text[1:] 

1032 elif not any(ch in text for ch in "\\{*?["): 

1033 retval = text 

1034 elif (node.type == "string" 

1035 and all(c.type in ('"', "string_content") for c in node.children)): 

1036 retval = _dq_unescape(text[1:-1]) 

1037 elif node.type == "concatenation" and _contiguous(node): 

1038 parts = [_literal_word(c, src) for c in node.children] 

1039 retval = "".join(parts) if all(p is not None for p in parts) else None 

1040 return retval

1043def _dq_unescape(body): 

1044 """The word a double-quoted string with no expansion in it becomes. Inside 

1045 double quotes a backslash escapes only `$`, a backquote, `"`, `\\` and a 

1046 newline (which it removes with itself); before anything else it stays, so 

1047 `"\\n"` is the two characters backslash and n, which is what curl's `-w` 

1048 and printf want. That is fixed by the shell, not by the command, so the 

1049 result is the word the command gets.""" 

1050 out, i = [], 0 

1051 while i < len(body): 

1052 ch = body[i] 

1053 if ch == "\\" and i + 1 < len(body): 

1054 nxt = body[i + 1] 

1055 if nxt in '$`"\\': 

1056 out.append(nxt) 

1057 elif nxt != "\n": 

1058 out.append(ch + nxt) 

1059 i += 2 

1060 else: 

1061 out.append(ch) 

1062 i += 1 

1063 retval = "".join(out) 

1064 return retval

1175# A heredoc is granted only in the shapes where bash and tree-sitter provably 

1176# agree on where it ends and what its text does: a plain delimiter or a wholly 

1177# quoted one, and a closing line that is the delimiter and nothing else. In an 

1178# unquoted body, anything the parser does not model is refused from the raw 

1179# bytes, because a node that is not in the tree cannot be walked. 

1180_HEREDOC_OPEN = frozenset(("<<", "<<-", "heredoc_start", "heredoc_body", "heredoc_end")) 

1181_HEREDOC_DELIM = re.compile(r"""[A-Za-z0-9_]+|'[A-Za-z0-9_.-]+'|"[A-Za-z0-9_.-]+\"""") 

1182# `$[` is arithmetic in its old spelling, code for the same reason `$((` is 

1183# (`_evaluates`); `$((` is caught by `$(`. 

1184_HEREDOC_UNSAFE = (b"`", b"$(", b"$[", b"\\\n")

1187def _heredoc_refusal(c, src): 

1188 """Why this heredoc's extent or body cannot be trusted, or None.""" 

1189 kinds = {k.type: k for k in c.children} 

1190 start = kinds.get("heredoc_start") 

1191 end = kinds.get("heredoc_end") 

1192 if start is None or end is None: 

1193 return "heredoc without an opening or closing delimiter" 

1194 raw = _text(start, src) 

1195 if not _HEREDOC_DELIM.fullmatch(raw): 

1196 return "heredoc delimiter is not plain or wholly quoted" 

1197 delim = _unquote(raw).encode() 

1198 clean = (lambda ln: ln.lstrip(b"\t")) if "<<-" in kinds else (lambda ln: ln) 

1199 last = src.rfind(b"\n", 0, end.start_byte) + 1 

1200 if clean(src[last:end.end_byte]) != delim or src[end.end_byte:end.end_byte + 1] not in (b"", b"\n"): 

1201 return "heredoc closing line is not its delimiter" 

1202 body = src[src.find(b"\n", start.end_byte) + 1:last] 

1203 if any(clean(ln) == delim for ln in body.split(b"\n")): 

1204 return "heredoc delimiter appears inside its body" 

1205 if raw == _unquote(raw) and any(tok in body for tok in _HEREDOC_UNSAFE): 

1206 return "command substitution in a heredoc or herestring" 

1207 return None

1280 def loop(self, node): 

1281 """A plain `for x in <words>; do <reads>; done` (#373). 

1282 

1283 Each part is allowlisted rather than the known-bad ones refused: the 

1284 keyword is `for` (tree-sitter parses `select` into the same node), the 

1285 list is words this layer can read without running anything, and the 

1286 body is walked like any command with only the read rules in force. 

1287 A C-style loop and `while` are other node types and still park.""" 

1288 kids = node.children 

1289 if not kids or kids[0].type != "for": 

1290 return (False, f"{kids[0].type if kids else 'loop'} loop is not grantable") 

1291 var = node.child_by_field_name("variable") 

1292 if var is None or not _LOOP_VAR.fullmatch(_text(var, self.src)): 

1293 return (False, "for: only a short lower-case loop variable is grantable") 

1294 if not any(c.type == "in" for c in kids): 

1295 # Without `in` it iterates "$@", which this layer cannot see. 

1296 return (False, "for: a loop without `in` is not grantable") 

1297 body = None 

1298 for c in kids[1:]: 

1299 if c.type in ("variable_name", "in", ";", "\n"): 

1300 continue 

1301 if c.type == "do_group": 

1302 body = c 

1303 elif node.field_name_for_child(list(kids).index(c)) == "value": 

1304 raw = _text(c, self.src) 

1305 # A quoted `$(` or backtick is inert in the list, but the 

1306 # guardrail is the text: nothing that looks like it runs. 

1307 if (not _static(c) or _glob_text(c, self.src) is None 

1308 or "$(" in raw or "`" in raw): 

1309 return (False, "for: the list is not literals or globs") 

1310 else: 

1311 return (False, f"for: {c.type} is not grantable") 

1312 if body is None: 

1313 return (False, "for: no body") 

1314 retval = (True, "") 

1315 for c in body.children: 

1316 if c.type in ("do", "done"): 

1317 continue 

1318 retval = self.read_scope(self.walk, c) 

1319 if not retval[0]: 

1320 break 

1321 return retval

1323 def substitutions(self, verb, args): 

1324 """None when every command substitution in args can be granted, else 

1325 why not (#373). 

1326 

1327 A substitution's output becomes argv, and a word in it that starts 

1328 with a dash becomes an option -- for `git grep`, the one that runs a 

1329 pager. So the rule is about where the words land, not what produced 

1330 them: after a `--` that is no option's value, and only for a command that 

1331 reads past that point as names and nothing else. READ_VERBS qualify by 

1332 their own bar (no argument makes them write). `git` qualifies through 

1333 its read rules, which still judge every word before the `--`. The 

1334 flag-checked verbs do not: `find`, `awk` and `date` read their 

1335 positionals as an expression, a program and a clock setting, so a 

1336 `--` does not make the output inert there. The inner command is walked 

1337 with only the read rules in force. Nothing here asks the classifier.""" 

1338 guarded = False 

1339 opaque = [] 

1340 prev = "" 

1341 for a in args: 

1342 if _no_substitution(a): 

1343 # git asks again after its subcommand, where its own `--` is. 

1344 word = _literal_word(a, self.src) if _static(a) else None 

1345 guarded = guarded or (word == _END_OF_OPTIONS and _frees_marker(prev)) 

1346 prev = word 

1347 continue 

1348 if verb not in READ_VERBS and verb != "git": 

1349 return "only a read verb or git takes one" 

1350 if not guarded: 

1351 return "no `--` before it that ends the options, so its output could be an option" 

1352 inner = _bare_substitution(a) 

1353 if inner is None: 

1354 return "not a whole argument" 

1355 # Its output is captured into argv, so a redirect on the command 

1356 # taking it is not where the inner command writes. 

1357 saved = (self.writes_file, self.heredoc_stdin) 

1358 self.writes_file = self.heredoc_stdin = False 

1359 try: 

1360 for c in inner.children: 

1361 if c.type in ("$(", ")", "`"): 

1362 continue 

1363 ok, why = self.read_scope(self.walk, c) 

1364 if not ok: 

1365 return f"inside it, {why}" 

1366 finally: 

1367 self.writes_file, self.heredoc_stdin = saved 

1368 opaque.append((a.start_byte, a.end_byte)) 

1369 self._opaque = frozenset(opaque) 

1370 return None

1423 def redirect_write(self, c): 

1424 """One file_redirect judged: (refusal, writes_a_file). 

1425 

1426 `refusal` is a reason string when the redirect cannot be judged at all, 

1427 and None otherwise -- in which case the second value says whether it 

1428 puts bytes in a file. 

1429 

1430 Shared by both callers on purpose. A redirect can sit on either side of 

1431 its command, and bash writes the file either way, so the two parse 

1432 shapes have to reach the same verdict or the rule is decoration (#213). 

1433 """ 

1434 # Words after the target belong to the command, whatever node the 

1435 # parser hangs them on, and nothing here reads them as arguments. 

1436 dests = c.children_by_field_name("destination") 

1437 if len(dests) > 1 or (dests and any(k.type in _CLOSE_OPS for k in c.children)): 

1438 return ("words after a redirect target are not grantable", None) 

1439 # No path, nothing to judge: not a device, not in the tree, not a 

1440 # write. Checked before the destination is read at all, because for 

1441 # `2>&1` the "destination" is the number 1. 

1442 if _fd_dup(c): 

1443 return (None, False) 

1444 dest = c.children[-1] if c.children else None 

1445 if dest is None or not _static(dest): 

1446 return ("redirect target is not a literal", None) 

1447 target = _literal_word(dest, self.src) 

1448 if target is None: 

1449 return ("redirect target is not a plain literal", None) 

1450 # `~+` and `~-` expand to directories `cd` chose, so only the home form 

1451 # can be read from here. 

1452 if target.startswith("~") and target != "~" and not target.startswith("~/"): 

1453 return ("redirect target is not a plain literal", None) 

1454 if self.tracked is None and not os.path.isabs(os.path.expanduser(target)): 

1455 return ("redirect target is relative to a directory this cannot read", None) 

1456 if _is_device(target, self.tracked) and target not in _DEV_OK: 

1457 return (f"redirect to device {target}", None) 

1458 # A read verb stops being a read when its output lands in a file 

1459 # (#177). YOLT cannot tell us this -- `cat x`, `cat x > out.txt` and 

1460 # `cat x > /usr/local/bin/foo` are one `unknown` to it, differing only 

1461 # in a reason string nobody parses -- so the redirect is seen here or 

1462 # not at all. 

1463 retval = (None, not _reads_only(c) and target not in _DEV_OK) 

1464 return retval

1466 def stdin_redirect(self, c): 

1467 """One heredoc or herestring judged: (refusal, writes_a_file, trailing). 

1468 

1469 Their text is expanded by bash exactly as an argument is, so it answers 

1470 to the same rule as an argument. What follows a heredoc's opening word 

1471 on its line hangs under this node: a redirect answers to 

1472 `redirect_write`, and the rest of a pipeline or list is returned for 

1473 the caller to walk after the command it follows, in source order.""" 

1474 trailing = [] 

1475 if not _no_substitution(c): 

1476 return ("command substitution in a heredoc or herestring", None, trailing) 

1477 writes = False 

1478 if c.type == "heredoc_redirect": 

1479 refusal = _heredoc_refusal(c, self.src) 

1480 if refusal is not None: 

1481 return (refusal, None, trailing) 

1482 for part in c.children: 

1483 if part.type == "file_redirect": 

1484 refusal, redirect_writes = self.redirect_write(part) 

1485 if refusal is not None: 

1486 return (refusal, None, trailing) 

1487 writes = writes or redirect_writes 

1488 elif part.type in ("command", "pipeline", "list", "redirected_statement"): 

1489 trailing.append(part) 

1490 elif part.type not in _HEREDOC_OPEN and part.type not in ("|", "&&", "||"): 

1491 return (f"{part.type} after a heredoc is not grantable", None, trailing) 

1492 return (None, writes, trailing)

1494 def redirected(self, node): 

1495 """A command, pipeline or list with redirects. The body is judged as 

1496 usual; the redirect is an in-tree write the sandbox confines, unless 

1497 it names a device other than the three harmless ones.""" 

1498 body = None 

1499 writes = False 

1500 trailing = [] 

1501 redirects = [] 

1502 for c in node.children: 

1503 if c.type in ("command", "pipeline", "list", "for_statement"): 

1504 body = c 

1505 elif c.type == "file_redirect": 

1506 refusal, redirect_writes = self.redirect_write(c) 

1507 if refusal is not None: 

1508 return (False, refusal) 

1509 writes = writes or redirect_writes 

1510 redirects.append(c) 

1511 elif c.type in ("heredoc_redirect", "herestring_redirect"): 

1512 refusal, redirect_writes, more = self.stdin_redirect(c) 

1513 if refusal is not None: 

1514 return (False, refusal) 

1515 writes = writes or redirect_writes 

1516 trailing.extend(more) 

1517 else: 

1518 return (False, f"{c.type} under a redirect is not grantable") 

1519 if body is None: 

1520 return (False, "redirect without a command") 

1521 if writes and (self.reads_only or body.type == "for_statement"): 

1522 # A loop or a substitution is granted as a read (#373); a file it 

1523 # writes would make it something else. `2>/dev/null`, `2>&1` and 

1524 # `<` write nothing and stay. 

1525 return (False, "a redirect to a file in a loop or a substitution") 

1526 before = len(self.reasons) 

1527 saved = self.writes_file 

1528 saved_heredoc = self.heredoc_stdin 

1529 # `cat > f <<EOF` is how an agent writes a file. The heredoc was judged 

1530 # above, so a lone `cat` reading it is the same write `tee f` is. 

1531 # Only when the heredoc is the sole stdin and stdout is what lands in 

1532 # the file: bash takes the last stdin redirect, so a `<` beside it 

1533 # would make this a file copy (#177), and a stderr redirect writes 

1534 # nothing of cat's output. 

1535 # A redirect written after the heredoc's opening word hangs under the 

1536 # heredoc node, so its children count too. 

1537 kids = list(node.children) + list(body.children) 

1538 kids += [k for c in node.children if c.type == "heredoc_redirect" for k in c.children] 

1539 self.heredoc_stdin = ( 

1540 body.type == "command" 

1541 and sum(c.type == "heredoc_redirect" for c in kids) == 1 

1542 and not any(c.type == "herestring_redirect" or c.type == "file_redirect" and _reads_only(c) 

1543 for c in kids) 

1544 and any(c.type == "file_redirect" and _stdout_write(c) for c in kids)) 

1545 self.writes_file = self.writes_file or writes 

1546 if writes: 

1547 # A redirect writes before a later segment runs, which is the same 

1548 # hazard the result path marks for `cp`/`mv` (#328 r2): 

1549 # `printf x > .vercel/project.json && vercel deploy` must park. 

1550 self.other_effect = True 

1551 retval = self.walk(body) 

1552 self.writes_file = saved 

1553 self.heredoc_stdin = saved_heredoc 

1554 if retval[0]: 

1555 # A redirect after a list opens its file once the list's `cd`s 

1556 # have run, so the directory is judged again where it now stands. 

1557 for c in redirects: 

1558 refusal, _ = self.redirect_write(c) 

1559 if refusal is not None: 

1560 return (False, refusal) 

1561 if retval[0] and len(self.reasons) > before: 

1562 self.reasons[-1] += " > redirect" 

1563 for part in trailing: 

1564 if not retval[0]: 

1565 break 

1566 retval = self.walk(part) 

1567 return retval

1595 def segment(self, node): 

1596 """A redirect may sit before its command as well as after it. 

1597 

1598 `>out.txt cat f` is `cat f > out.txt` with the words in the other 

1599 order, and bash writes the file for both. Only the trailing form parses 

1600 as a `redirected_statement`; the leading one is a `file_redirect` child 

1601 of the command node, which `redirected()` never sees and `argv()` 

1602 skips outright -- so `cat f > out.txt` parked while `>out.txt cat f` 

1603 was granted as "cat: read-only", and the difference was word order 

1604 (#213). Found reviewing the descriptor-dup change above; it predates 

1605 it.""" 

1606 saved = (self.writes_file, self.reads_only, self.unattended) 

1607 try: 

1608 retval = self._segment(node) 

1609 finally: 

1610 self.writes_file, self.reads_only, self.unattended = saved 

1611 return retval

1615 # Kept for the handlers that have to ask about reach: an unattended 

1616 # `git push` is granted on its subcommand and still has to name a host 

1617 # this channel was given (#253). 

1618 self._seg_text = text 

1619 self._opaque = frozenset() 

1620 for c in node.children: 

1621 if c.type == "file_redirect": 

1622 refusal, writes = self.redirect_write(c) 

1623 if refusal is not None: 

1624 return (False, refusal) 

1625 if writes and self.reads_only: 

1626 return (False, "a redirect to a file in a loop or a substitution")

1628 if writes: 

1629 self.other_effect = True 

1630 elif c.type in ("heredoc_redirect", "herestring_redirect"):

1632 if refusal is not None:

1634 if more:

1649 # `cp $(rm -rf x) b`, `cd $(curl ... | sh)`. The allowlisted verbs 

1650 # and `cd` are judged on the verb alone, and YOLT only sees the

1681 # be an aide there. So `rm -rf build`, `git push --force` and 

1682 # `gh pr merge` run -- `git` and `gh` still answer to

1687 # does not describe: `git push` to a remote outside `allow_domains` 

1688 # leaves the blast radius the operator drew, and leaving it is what 

1689 # a card is still for.

1700 elif verb == "aws": 

1701 retval = self.aws(args) 

1702 elif verb == "npm": 

1703 retval = self.npm(args)

1801 def awk_words(self, args, texts, at): 

1802 """Why awk's words are not ones whose program this layer can read, or 

1803 None. `at` is where the program is. 

1804 

1805 The options are data -- a separator, a variable -- so they are read

1809 

1810 The program is code, and its text must be exactly the word awk gets. 

1811 Inside single quotes the shell changes nothing; a double-quoted or 

1812 concatenated word, or a bare word, it can. The operands after it keep 

1813 the rule every awk word had before #374: a plain word or a

1822 elif not all(a.type == "raw_string" 

1823 or a.type in ("word", "number") and "\\" not in _text(a, self.src) 

1824 for a in args[at + 1:]): 

1825 retval = "awk: an operand is not a plain word or single-quoted string" 

1826 elif any(t.startswith(("/inet", "/p/inet")) for t in texts[at + 1:]): 

1827 retval = "awk: an operand names a gawk network file" 

1828 return retval

1830 def fetch(self, verb, text, args): 

1831 """A `curl` to a host the channel has already allow-listed (#239). 

1832 

1833 The credential half of a channel and the network half never met: a 

1834 policy could hold `api.example.com` in `allow_domains` and the token to 

1835 use it in `env`, and every `curl` still parked, because this layer had 

1836 no rule for fetch verbs and voitta-yolt 2.0.x delegates them here. The 

1837 tool that does honour `allow_domains` -- `web_fetch` -- sends no 

1838 headers, so the authenticated read had no uncarded path at all. 

1839 

1840 Nothing about what may be reached moves: `check_egress` is the same 

1841 function, with the same answers, and it already refuses a host that is 

1842 not statically visible, a host outside the list, and the upload-shaped 

1843 flags #222 added. What moves is that meeting those conditions is now a 

1844 grant rather than a fall-through to a classifier that will punt. 

1845 

1846 `wget` is deliberately not here. It writes the response to a file by 

1847 default, so the no-output-flag rule below would have to be inverted for

1853 if word is not None: 

1854 texts.append(word) 

1855 elif _static(a): 

1856 retval = (False, f"{verb}: an argument takes shell rules to read") 

1857 return retval 

1858 elif _quoted_non_flag(raw):

1860 # where the credential comes from the channel's `env` and never 

1861 # appears in argv. Inside quotes an expansion cannot split into 

1862 # further arguments, and a literal first character that is not 

1863 # a dash cannot become an option -- so this word is a value, 

1864 # whatever it expands to, and there is no flag here to read.

1866 else: 

1867 retval = (False, f"{verb}: an argument is not a literal, so its flags cannot be read") 

1868 return retval 

1869 # Reach first, shape second. Both refuse, but `curl -sXPOST` carries an 

1870 # option letter inside its attached value ("POST" holds an O), and the 

1871 # cluster scan below cannot tell that from `-sO`. Asking check_egress 

1872 # first means such a command is refused for carrying a request body, 

1873 # which is what it does, rather than for a file it does not write. 

1874 allowed, why = policy_mod.check_egress(text, self.policy) 

1875 if not allowed: 

1876 retval = (False, why) 

1877 return retval 

1878 hit = _fetch_write_flag(texts) 

1879 if hit and self.unattended and _fetch_names_its_file(texts): 

1880 # An unattended channel already writes its tree without a card 

1881 # (#253), and the sandbox confines this write the same way. Only 

1882 # `-o FILE`: a name taken from the URL or a response header is 

1883 # still the server choosing it.

1885 return retval 

1886 if hit: 

1887 # A fetch that also writes a file is two powers in one command, and 

1888 # the file is the response body -- content from off the box landing 

1889 # in the channel's tree under a name the command chose. That can 

1890 # have its own card; reading to stdout is what this grants. 

1891 retval = (False, f"{verb}: {hit} writes the response to a file") 

1892 return retval 

1893 retval = (True, f"{verb}: fetch to a host in allow_domains") 

1894 return retval

1897 """Move the tracked directory, or make it unknown wherever bash could

2456def _git_pin_refusal(directory, start_dir): 

2457 """Why git may not be granted in `directory`, or None. 

2458 

2459 A git read is only as safe as the repository configuration it loads, and 

2460 that configuration can name commands git runs. So git is granted only where 

2461 the channel's policy put a repository: its checkout, or a worktree root 

2462 under the `<checkout>.worktrees` sibling the sandbox makes writable for 

2463 that purpose. Anywhere else -- a subdirectory, the temp dir, a parent -- 

2464 the repository git would discover is one nobody named.""" 

2465 if directory is None: 

2466 return "git: directory not statically known" 

2467 d = os.path.realpath(directory) 

2468 root = os.path.realpath(start_dir) 

2469 # Built the way sandbox.roots builds the writable sibling. 

2470 worktrees = os.path.realpath(root + ".worktrees") 

2471 dot = os.path.join(d, ".git") 

2472 # A `.git` symlink hands the repository to wherever it points. 

2473 if os.path.islink(dot): 

2474 return "git: the repository's .git is a symlink" 

2475 if d == root and os.path.isdir(dot): 

2476 return None 

2477 # A checkout that is itself a linked worktree has nothing trusted to hold 

2478 # its pointer to: a gitdir of the right shape can be built anywhere the 

2479 # sandbox writes, by renaming a directory into place (#344 review). 

2480 if d == root and os.path.isfile(dot): 

2481 return "git: the checkout's .git is a file, with no repository to check it against" 

2482 if d.startswith(worktrees + os.sep) and os.path.isfile(dot): 

2483 main = os.path.join(root, ".git") 

2484 if os.path.islink(main) or not os.path.isdir(main): 

2485 return "git: the checkout's .git is not a directory" 

2486 retval = _pointer_refusal(dot, os.path.realpath(main)) 

2487 return retval 

2488 retval = "git: not the channel's checkout or one of its worktrees" 

2489 return retval

2492def _pointer_refusal(dot, main): 

2493 """Why a linked worktree's `.git` file may not be trusted, or None. 

2494 

2495 The file names the gitdir git loads, and both it and the gitdir's 

2496 `commondir` are writable from inside the sandbox: a forged pair can aim 

2497 git at a bare repository in the temp dir whose `config` sets 

2498 `core.fsmonitor`, and the probes run git outside the sandbox (#344 

2499 review). So the pointer has to have the shape `git worktree add` gives it: 

2500 a gitdir directly under `<C>/worktrees/`, whose `commondir` resolves back 

2501 to `<C>`, where `<C>` is `main`, the channel's own checkout's `.git`.""" 

2502 gitdir = _read_pointer(dot, "gitdir: ") 

2503 if gitdir is None: 

2504 return "git: the worktree's .git is not a single gitdir line" 

2505 gitdir = os.path.realpath(os.path.join(os.path.dirname(dot), gitdir)) 

2506 admin = os.path.dirname(gitdir) 

2507 common = os.path.dirname(admin) 

2508 if (os.path.basename(admin) != "worktrees" or os.path.basename(common) != ".git" 

2509 or common != main or not os.path.isdir(gitdir)): 

2510 return "git: the worktree's .git does not point into a repository's .git/worktrees" 

2511 pointed = _read_pointer(os.path.join(gitdir, "commondir"), "") 

2512 if pointed is None or os.path.realpath(os.path.join(gitdir, pointed)) != common: 

2513 return "git: the worktree's commondir is not the repository it belongs to" 

2514 retval = None 

2515 return retval

2518def _read_pointer(path, prefix): 

2519 """The one line of a small git pointer file after `prefix`, or None.""" 

2520 try: 

2521 with open(path, encoding="utf-8") as f: 

2522 text = f.read(4097) 

2523 except (OSError, UnicodeDecodeError): 

2524 return None 

2525 lines = text.splitlines() 

2526 if len(text) > 4096 or len(lines) != 1 or not lines[0].startswith(prefix): 

2527 return None 

2528 retval = lines[0][len(prefix):] or None 

2529 return retval

2553def _vercel_deployment_project(target, names, team): 

2554 """The allowed project a deployment URL belongs to, or None. 

2555 

2556 Only Vercel's generated `<project>-<hash>-<team>.vercel.app` is read. A 

2557 custom domain, a `dpl_` ID or a branch URL names no project unambiguously 

2558 in its text, so it is unresolvable and parks.""" 

2559 host = target.split("://", 1)[-1].split("/", 1)[0].lower() 

2560 suffix = ".vercel.app" 

2561 if not team or not host.endswith(suffix): 

2562 return None 

2563 label = host[:-len(suffix)] 

2564 tail = "-" + str(team).lower() 

2565 if not label.endswith(tail): 

2566 return None 

2567 label = label[:-len(tail)] 

2568 # Exactly `<name>-<9-char hash>`. A prefix match let `site` claim 

2569 # `site-evil-<hash>`, a different project's deployment. The 

2570 # `<name>-git-<branch>` form is not accepted: `site-git-x` is equally a 

2571 # project named `site-git` deploying branch `x`, and nothing in the text 

2572 # tells them apart. 

2573 m = _VERCEL_HASHED.match(label) 

2574 if m and m.group(1) in names: 

2575 return m.group(1) 

2576 return None
