1"""Learning L0 (#129): the agent flags, a trusted user decides, a PR is the 

2proposal, a merge is the promotion. 

3 

4Design in #100 and #52 -- learning inherits the authz spine. What that means 

5here: 

6 

7- flag_skill is the one tool the model may call on its own initiative. It 

8 parks a proposal (proposals.py) and the ingest posts a card tagging the 

9 trusted users. Nothing is drafted or written by a flag. 

10- propose_skill / decline_skill act on a parked proposal and are trusted-only, 

11 like approve_command. propose_skill reads the thread's trajectory back 

12 (trajectory.py), asks the waterfall for a SKILL.md in the skillz format, and 

13 opens a PR against `learning.repo` under `learning.path` -- through `gh api` 

14 from THIS process, never from the channel's shell: the sandbox (#116) 

15 confines a channel to its tree, and the model must not gain a push path to 

16 the skills repo. The PR is the record; a human merging it is the promotion. 

17- One flag per thread, and a declined or proposed thread is not asked again 

18 (state.py, keyed on thread ts), so the card cannot become a nag. 

19 

20A skill loaded later carries no authority: it is prompt text, and whatever it 

21does still goes through run_shell -> YOLT -> grant -> sandbox -> approval."""

36# description, and as a block in the system prompt where it is in view while the 

37# answer is being composed. Two copies that must agree is how they stop agreeing,

39#

42# composition, so a judgment about the turn's conclusion is being asked for 

43# through a mechanism that runs before the conclusion exists. The bar cannot 

44# create the missing hook, but it can stop describing the wrong subject -- the 

45# old text named three shapes, all of them things you DO, and the turn it missed 

46# was one where the agent mostly read three facts and thought.

49 "trusted user can decide. At most once per thread, at the end of a turn.\n" 

50 "Judge the turn's CONCLUSION, not the tool calls that produced it. A turn " 

51 "that ran no commands and reasoned its way to a general rule can meet the " 

52 "bar; a turn that ran ten and answered a routine question does not.\n"

55 "- a workaround found by trial and error;\n"

168def classify_scope(name, why, channel): 

169 """(scope, reason). Which catalog this belongs in, and why -- the reason is 

170 shown on the card, because a proposal a human cannot check is a proposal 

171 they have to take on faith.""" 

172 hay = f"{name} {why}" 

173 hit = _PRIVATE_HINTS.search(hay) 

174 if hit: 

175 retval = ("channel", f"names something specific to here ({hit.group(0)[:40]})") 

176 return retval 

177 slug = channel_slug(channel) 

178 if slug and slug in hay.lower(): 

179 retval = ("channel", f"names this channel ({slug})") 

180 return retval 

181 retval = ("shared", "nothing channel-specific found in the name or reason") 

182 return retval

185def amend_candidate(name, why, channel): 

186 """An existing skill this might amend, or None (#210). 

187 

188 The first real run drafted a sibling of a public skill that already covered 

189 the same ground, because nothing looked. This searches what this instance 

190 can actually see -- the loaded index -- and matches on the name's words. It 

191 does NOT reach the public skillz catalog: that needs a network call from the 

192 host process, and a missed duplicate costs a review comment while a new 

193 fetch path costs a review of its own.""" 

194 words = {w for w in name.split("-") if len(w) > 3} 

195 if not words: 

196 return None 

197 best, score = None, 0 

198 for entry in skills.view(channel).values(): 

199 hay = f"{entry.get('name', '')} {entry.get('summary', '')}".lower() 

200 n = sum(1 for w in words if w in hay) 

201 if n > score: 

202 best, score = entry, n 

203 # Two shared significant words is the bar. One is a coincidence ("github"), 

204 # and requiring the whole name back would only ever match itself. 

205 retval = best if score >= 2 else None 

206 return retval

224def flag(args, ctx): 

225 """The model's flag. Parks a proposal for the ingest to render; refuses a 

226 second flag on a thread that already has one, or was declined.""" 

227 if not enabled(): 

228 retval = "learning is not configured on this instance (learning.repo); nothing flagged." 

229 return retval 

230 channel, thread_ts = ctx.get("channel"), ctx.get("thread_ts") 

231 if not (channel and thread_ts): 

232 retval = "flag_skill needs a channel thread; nothing flagged." 

233 return retval 

234 prior = thread_state(thread_ts) 

235 if prior: 

236 retval = f"this thread was already {prior} as a skill candidate; not asking again." 

237 return retval 

238 name = _slug(args.get("name")) 

239 why = " ".join(str(args.get("why") or "").split())[:300] 

240 scope, reason = classify_scope(name, why, channel) 

241 amend = amend_candidate(name, why, channel) 

242 key = proposals.add(name, why, channel, thread_ts, ctx.get("user_id"), 

243 scope=scope, scope_reason=reason, 

244 amends=(amend or {}).get("name")) 

245 mark_thread(thread_ts, "flagged") 

246 retval = ( 

247 f"flagged [{key}] `{name}` -- a card tagging the trusted users will follow " 

248 f"this reply. Do not draft the skill yourself; a trusted user opens the PR or declines." 

249 ) 

250 return retval

286def _parse_verdict(text): 

287 """(name, why) to flag, or (None, reason) not to. A reply that fits neither 

288 shape is treated as 'no', with the raw line as the reason, so a confused 

289 model never produces a card.""" 

290 first = (text or "").strip().splitlines() 

291 line = first[0] if first else "" 

292 if line[:5].upper() == "FLAG:": 

293 name, _, why = line[5:].strip().partition("|") 

294 name = name.strip() 

295 if name: 

296 retval = (name, why.strip() or "flagged by the post-answer check") 

297 return retval 

298 if line[:3].upper() == "NO:": 

299 retval = (None, line[3:].strip() or "no reason given") 

300 return retval 

301 retval = (None, f"unparsed verdict: {line[:160]}") 

302 return retval

305def check(answer, trace, channel, thread_ts, user_id): 

306 """Post-answer skill check (#232). Returns a short string for the trajectory's 

307 flag_considered field -- 'flagged (...)' or 'considered: <reason>' -- or None 

308 when the check did not run. Never raises and never blocks the reply: it runs 

309 after the answer is composed. 

310 

311 On FLAG it parks the proposal through flag(), exactly as the in-composition 

312 tool would, so the ingest surfaces the same card. On NO it only records the 

313 reason. The caller decides whether to run it at all (the per-channel switch); 

314 this still self-guards on enabled(), a live thread, and not-already-flagged.""" 

315 if not (enabled() and channel and thread_ts): 

316 retval = None 

317 return retval 

318 if thread_state(thread_ts): 

319 retval = None # already flagged or declined; flag() would refuse anyway 

320 return retval 

321 if not _candidate(trace): 

322 retval = None 

323 return retval 

324 steps_desc = "; ".join( 

325 f"{s.get('tool')}[{s.get('disposition')}]" for s in (trace or []) 

326 ) or "(no tool calls)" 

327 user = ( 

328 f"The turn's steps: {steps_desc}\n\n" 

329 f"The turn's final answer:\n{(answer or '').strip()[:4000]}" 

330 ) 

331 try: 

332 msg = llm.complete([ 

333 {"role": "system", "content": _CHECK_SYSTEM}, 

334 {"role": "user", "content": user}, 

335 ]) 

336 verdict = msg.content or "" 

337 except Exception as exc: 

338 logging.warning("learning: post-answer check failed, not flagging: %s", exc) 

339 retval = "considered: check call failed" 

340 return retval 

341 name, why = _parse_verdict(verdict) 

342 if name is None: 

343 logging.info("learning: post-answer check declined in %s (%s)", channel, why) 

344 retval = f"considered: {why}" 

345 return retval 

346 out = flag({"name": name, "why": why}, 

347 {"channel": channel, "thread_ts": thread_ts, "user_id": user_id}) 

348 logging.info("learning: post-answer check flagged %s in %s", name, channel) 

349 retval = f"flagged ({out})" 

350 return retval

381def _skill_text(raw): 

382 """The SKILL.md inside a model's answer, or None. Models wrap it despite 

383 being told not to: a sentence first ("Here is the skill:"), a ```markdown 

384 fence. So this looks for the first `---` line that opens frontmatter with 

385 a `name:` in it, and drops the fence that wrapped it, if one did, along 

386 with anything after that fence. A note after an UNfenced draft cannot be 

387 told from the skill's last paragraph and stays in; the draft becomes a 

388 pull request a person reviews, which is where it gets seen (#425 review).""" 

389 lines = raw.splitlines() 

390 for i, line in enumerate(lines): 

391 if line.strip() != "---": 

392 continue 

393 close = next((j for j in range(i + 1, min(len(lines), i + 40)) if lines[j].strip() == "---"), None) 

394 if close is None or not any(l.startswith("name:") for l in lines[i + 1:close]): 

395 continue 

396 # Trim a wrapping fence only: a skill's own code blocks are content. 

397 # The wrapper opened just above the frontmatter and closes last. 

398 prev = next((lines[k].strip() for k in range(i - 1, -1, -1) if lines[k].strip()), "") 

399 end = len(lines) 

400 if prev.startswith("```"): 

401 end = next((j for j in range(len(lines) - 1, close, -1) if lines[j].strip() == "```"), len(lines)) 

402 retval = "\n".join(lines[i:end]).strip() + "\n" 

403 return retval 

404 return None

407def draft(name, why, channel, thread_ts): 

408 """A SKILL.md from the thread's trajectory, via the waterfall. Returns 

409 (text, error).""" 

410 records = trajectory.thread(channel, thread_ts) 

411 if not records: 

412 retval = (None, "no trajectory records for this thread") 

413 return retval 

414 today = datetime.date.today().isoformat() 

415 payload = json.dumps(records, ensure_ascii=False)[:60000] 

416 messages = [ 

417 {"role": "system", "content": _DRAFT_SYSTEM}, 

418 {"role": "user", "content": ( 

419 f"Skill name: {name}\nWhy it is worth a skill: {why}\nDate: {today}\n" 

420 f"Channel: {channel_slug(channel)}\n\nTurn records (JSON, oldest first):\n{payload}" 

421 )}, 

422 ] 

423 try: 

424 text = (llm.complete(messages).content or "").strip() 

425 except Exception: # noqa: BLE001 

426 # The error goes to the log only (#351, #352): a router failure's text 

427 # is the whole multi-provider dump, and this line lands in the channel. 

428 logging.exception("learning: drafting %s failed", name) 

429 retval = (None, "the language model call failed; the details are in my log") 

430 return retval 

431 found = _skill_text(text) 

432 if found is None: 

433 logging.warning("learning: draft for %s had no frontmatter; it began %r", 

434 name, redact.scrub(text[:200])) 

435 retval = (None, "draft did not start with frontmatter") 

436 return retval 

437 text = found 

438 retval = (redact.scrub(text), None) 

439 return retval

442def _gh(method, path, payload=None): 

443 """One GitHub API call through gh, which holds the operator's token in the 

444 keychain. Raises RuntimeError with gh's (scrubbed) stderr on failure.""" 

445 argv = ["gh", "api", "--method", method, path] 

446 stdin = None 

447 if payload is not None: 

448 argv += ["--input", "-"] 

449 stdin = json.dumps(payload) 

450 proc = subprocess.run(argv, input=stdin, capture_output=True, text=True, timeout=60) 

451 if proc.returncode != 0: 

452 raise RuntimeError(f"gh api {method} {path}: {redact.scrub(proc.stderr.strip())[:300]}") 

453 retval = json.loads(proc.stdout) if proc.stdout.strip() else {} 

454 return retval

471def open_pr(key, name, channel, text, body, api=_gh, scope="channel"): 

472 """Branch, file and PR in learning.repo. Returns the PR URL. 

473 

474 Resumable, keyed on the proposal id: a retry after a failure part-way 

475 (branch made, file not; file made, PR not) finds what exists and 

476 continues, so a transient error never strands a half-made proposal or 

477 makes a second branch for the same one.""" 

478 repo, base = config.LEARNING_REPO, config.LEARNING_BASE 

479 cslug = channel_slug(channel) 

480 # Scope picks the destination inside the same catalog (#210): the 

481 # per-channel path, or the same template with the channel segment removed. 

482 template = config.LEARNING_PATH if scope == "channel" else shared_path() 

483 path = template.format(channel=cslug, name=name) 

484 # The branch keeps the channel either way -- it is provenance, not 

485 # destination, and two channels learning the same shared skill must not 

486 # collide on one branch name. 

487 branch = f"skill/{cslug}/{name}-{proposals.canonical(key)}" 

488 if _optional(api, f"repos/{repo}/git/ref/heads/{branch}") is None: 

489 sha = api("GET", f"repos/{repo}/git/ref/heads/{base}")["object"]["sha"] 

490 api("POST", f"repos/{repo}/git/refs", {"ref": f"refs/heads/{branch}", "sha": sha}) 

491 existing = _optional(api, f"repos/{repo}/contents/{path}?ref={branch}") 

492 put = { 

493 "message": f"skill: {name} ({cslug})", 

494 "content": base64.b64encode(text.encode("utf-8")).decode("ascii"), 

495 "branch": branch, 

496 } 

497 if existing and existing.get("sha"): 

498 put["sha"] = existing["sha"] 

499 api("PUT", f"repos/{repo}/contents/{path}", put) 

500 owner = repo.split("/", 1)[0] 

501 open_prs = _optional(api, f"repos/{repo}/pulls?head={owner}:{branch}&state=open") or [] 

502 if open_prs: 

503 retval = open_prs[0].get("html_url", "") 

504 return retval 

505 pr = api("POST", f"repos/{repo}/pulls", { 

506 "title": f"skill: {name} ({cslug})", "head": branch, "base": base, "body": body, 

507 }) 

508 retval = pr.get("html_url", "") 

509 return retval

520def propose(key, ctx, api=_gh, scope=None): 

521 """A trusted user said yes, by text: acquire, then the acquired core. 

522 Trust is the caller's check (admin_tools), the same as approve_command. 

523 

524 `scope` overrides what the classifier proposed at flag time (#210). The 

525 classifier only ever proposes; this is the override, and it arrives in 

526 words ("open it for every channel") rather than as another button, so the 

527 click path that approvals and proposals share does not grow a third 

528 action.""" 

529 channel = ctx.get("channel") 

530 k = proposals.canonical(key) 

531 prop = proposals.acquire(k, channel) 

532 if prop is None: 

533 if proposals.status(k, channel)[0] == "held": 

534 retval = (f"[{k}] is already being acted on by another surface -- do not " 

535 f"retry; that surface will report the outcome.") 

536 return retval 

537 parked = len(proposals.ids(channel)) 

538 also = f" {parked} other proposal(s) are parked here." if parked else "" 

539 retval = f"no pending skill proposal '{k}' in this channel.{also}" 

540 return retval 

541 try: 

542 retval = propose_acquired(k, prop, ctx, api=api, scope=scope) 

543 finally: 

544 proposals.release(k) 

545 return retval

548def propose_acquired(key, prop, ctx, api=_gh, scope=None): 

549 """The work, for a proposal the caller owns (#105). On a transient failure 

550 the proposal goes back under the SAME id via restore(); on success it is 

551 consumed with finish().""" 

552 channel = ctx.get("channel") 

553 name, why, thread_ts = prop["name"], prop["why"], prop["thread_ts"] 

554 # What the classifier proposed, unless a trusted user said otherwise. An 

555 # unrecognised value is ignored rather than guessed at: it would pick a 

556 # destination nobody asked for, and the proposed one is on the card. 

557 _scope = scope if scope in SCOPES else prop.get("scope", "channel") 

558 # Every failure below is one that may not repeat -- the waterfall was 

559 # down, the record file was not there yet, GitHub blinked -- so none of 

560 # them is a decline. The proposal goes back under the SAME id and the 

561 # card is re-rendered with its buttons; only a trusted user's Decline 

562 # closes a thread. 

563 text, err = draft(name, why, channel, thread_ts) 

564 if err: 

565 proposals.restore(key, prop) 

566 retval = f"{RETRY} could not draft `{name}`: {err}. Nothing was written; the proposal is still open." 

567 return retval 

568 meta, _body = skills._parse_text(text) 

569 if meta is None or not meta.get("name") or not meta.get("description"): 

570 proposals.restore(key, prop) 

571 retval = f"{RETRY} the draft for `{name}` had no usable frontmatter; nothing was written, the proposal is still open." 

572 return retval 

573 # The scope was decided from the flag's ONE LINE; the file is what actually 

574 # lands. Re-run the classifier on the draft, because the name and the reason 

575 # can be generic while the body is not -- which is the whole shape of #52's 

576 # envelope leak, one level in. 

577 # 

578 # When they disagree in the unsafe direction, re-park rather than pick. 

579 # Choosing silently would either ship a channel-specific skill to every 

580 # channel, or overrule a trusted user who asked for `shared`. Re-parking 

581 # hands them the new fact and keeps the decision theirs -- the same move 

582 # this function already makes for a draft with no usable frontmatter. 

583 # 

584 # An explicit override skips this: they have now been told once, and a 

585 # second refusal on the same grounds would be a loop rather than a warning. 

586 _draft_scope, _draft_reason = classify_scope(name, text, channel) 

587 if scope not in SCOPES and _scope == "shared" and _draft_scope == "channel": 

588 proposals.restore(key, {**prop, "scope": "channel", "scope_reason": _draft_reason}) 

589 retval = (f"{RETRY} the draft for `{name}` {_draft_reason}, which the one-line flag " 

590 f"did not show. Nothing was written. Re-parked as *this channel only* -- " 

591 f"approve again to take that, or ask for `shared` explicitly to override " 

592 f"now that the draft has been read.") 

593 return retval 

594 

595 link = _permalink({**ctx, "thread_ts": thread_ts}) 

596 body = ( 

597 f"Proposed from a shmobster turn in `#{channel_slug(channel)}`, flagged by the agent " 

598 f"and opened by {identity.mention(ctx.get('user_id'))}.\n\n**Why:** {why}\n\n" 

599 + (f"**Thread:** {link}\n\n" if link else "") 

600 + (f"**Scope:** {'this channel only' if _scope == 'channel' else 'every channel'}" 

601 + (f" -- {prop.get('scope_reason')}" if prop.get("scope_reason") else "") 

602 + (f", overridden by {identity.mention(ctx.get('user_id'))}" if scope in SCOPES else "") + "\n\n") 

603 + (f"**May amend:** `{prop['amends']}` -- check whether this belongs as a section there " 

604 f"rather than as a new file.\n\n" if prop.get("amends") else "") 

605 + "Merging this PR is what makes the skill load (learning L1, #130); until then it is a proposal.\n\n" 

606 "Provenance: shmobster #129." 

607 ) 

608 try: 

609 url = open_pr(key, name, channel, text, redact.scrub(body), api=api, scope=_scope) 

610 except Exception as exc: # noqa: BLE001 

611 logging.exception("learning: could not open the PR for %s", name) 

612 proposals.restore(key, prop) 

613 retval = f"{RETRY} could not open the PR for `{name}`: {redact.scrub(str(exc))[:300]}. The proposal is still open; a retry resumes where this stopped." 

614 return retval 

615 proposals.finish(key) 

616 mark_thread(thread_ts, "proposed") 

617 logging.info("learning: PR opened for %s in %s by %s: %s", name, channel, ctx.get("user_id"), url) 

618 _where = "this channel only" if _scope == "channel" else "every channel" 

619 retval = (f"PR opened by {identity.mention(ctx.get('user_id'))} for `{name}` ({_where}): {url}\n" 

620 f"Merging it is the promotion; nothing loads until then.") 

621 return retval

624def decline_acquired(key, prop, ctx): 

625 proposals.finish(key) 

626 mark_thread(prop["thread_ts"], "declined") 

627 retval = f"DECLINED by {identity.mention(ctx.get('user_id'))}: `{prop['name']}` -- this thread will not be asked again." 

628 return retval

631def decline(key, ctx): 

632 channel = ctx.get("channel") 

633 k = proposals.canonical(key) 

634 prop = proposals.acquire(k, channel) 

635 if prop is None: 

636 if proposals.status(k, channel)[0] == "held": 

637 retval = f"[{k}] is already being acted on by another surface -- do not retry." 

638 return retval 

639 retval = f"no pending skill proposal '{k}' in this channel." 

640 return retval 

641 try: 

642 retval = decline_acquired(k, prop, ctx) 

643 finally: 

644 proposals.release(k) 

645 return retval
