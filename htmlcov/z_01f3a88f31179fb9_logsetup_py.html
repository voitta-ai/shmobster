19class _PrivateRotatingFileHandler(logging.handlers.RotatingFileHandler): 

20 """RotatingFileHandler that keeps every file it opens at 0600. 

21 

22 A single chmod after construction is not enough, and the failure is 

23 invisible: each rollover opens a NEW file under the process umask, so with 

24 a 0022 umask the log is 0600 until it first rotates and 0644 forever after 

25 -- measured, all three files, while the one assertion that existed only 

26 looked at the first. Overriding _open covers the initial open and every 

27 rollover; an archive (.1, .2) inherits its mode from the file it was 

28 renamed from, which was opened here too.""" 

29 

30 def _open(self): 

31 retval = super()._open() 

32 os.chmod(self.baseFilename, 0o600) 

33 return retval

37 """The agent's own rotating log when config.logging.path is set (#155). 

38 

39 Without it, logging goes to stderr and the supervisor decides everything: 

40 the live deployment's launchd-redirected error log reached 185 MB of mostly 

41 reconnect noise, mode 0644 on a multi-user box, with nothing to rotate it. 

42 The contents are redacted at emission, so what leaks is not credentials -- 

43 it is every command every channel asked for, readable by any local account. 

44 

45 The directory is created 0700 and every file the handler opens is forced to 

46 0600 (see _PrivateRotatingFileHandler): the stdlib honors the process umask, 

47 which under launchd is whatever the plist says, and the sample plist cannot 

48 fix a file that already exists.""" 

49 if not config.LOG_PATH: 

50 retval = logging.StreamHandler() 

51 return retval 

52 directory = os.path.dirname(os.path.abspath(config.LOG_PATH)) 

53 os.makedirs(directory, mode=0o700, exist_ok=True) 

54 retval = _PrivateRotatingFileHandler( 

55 config.LOG_PATH, maxBytes=config.LOG_MAX_BYTES, backupCount=config.LOG_BACKUPS, 

56 ) 

57 return retval

65def warnings(): 

66 """What is wrong with where this process's log is actually going (#230). 

67 

68 Two things, and the second is the one that cost 251 MB. 

69 

70 **The managed log is opt-in and silently off.** Without `logging.path`, 

71 handler() returns a StreamHandler and everything goes to stderr, where the 

72 supervisor decides the mode and nothing rotates -- which is the exact 

73 failure #155 was written to fix, still reachable by leaving one key out. 

74 

75 **The supervisor's own file exists either way.** launchd redirects stderr 

76 to StandardErrorPath whatever the config says, and the crash that matters 

77 happens in slack_app's App() constructor at *import*, before main() ever 

78 calls setup(). So a crash-looping box writes the same traceback into an 

79 unrotated file forever, and configuring logging.path does not stop it. 

80 

81 Found by fstat on our own stderr rather than by reading the plist: the 

82 process does not know its StandardErrorPath, but it can ask what fd 2 is. 

83 That covers launchd, nohup, `2>file` and anything else, and says nothing 

84 when stderr is a tty or a pipe.""" 

85 retval = [] 

86 if not config.LOG_PATH: 

87 retval.append( 

88 "logging.path is unset, so the agent logs to stderr and whatever " 

89 "started it owns the file -- unrotated, at a mode this process did " 

90 "not choose. Set logging.path (see the example config) to get the " 

91 "rotating 0600 log (#155)" 

92 ) 

93 try: 

94 st = os.fstat(2) 

95 except OSError: 

96 st = None 

97 if st is not None and stat.S_ISREG(st.st_mode): 

98 if st.st_mode & 0o077: 

99 retval.append( 

100 f"stderr is a file with mode {st.st_mode & 0o777:04o} -- readable " 

101 "beyond its owner. It holds every command every channel asked for" 

102 ) 

103 if st.st_size > _STDERR_BIG: 

104 retval.append( 

105 f"stderr is a file of {st.st_size // (1024 * 1024)} MB and nothing " 

106 "here rotates it -- the supervisor redirected it (launchd's " 

107 "StandardErrorPath), so rotation is the supervisor's job " 

108 "(newsyslog.d) or the file needs truncating" 

109 ) 

110 return retval
