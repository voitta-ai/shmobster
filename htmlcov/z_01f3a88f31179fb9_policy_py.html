16def resolve(channel): 

17 # A gateway conversation linked to a Slack channel shares that channel's 

18 # policy (#336): one capability envelope, set in one place. 

19 # `link` and `mirror` are routing, not envelope. A `link` is followed first, 

20 # to the linked channel's policy; an entry left with nothing else (mirror 

21 # only) has no policy of its own and falls back to default_policy, as an 

22 # absent one does. Read truthily, `{"mirror": false}` would be an empty 

23 # envelope -- no github_repos bound at all -- which is wider than the default. 

24 channel = linked(channel) or channel 

25 own = {k: v for k, v in (config.CHANNEL_POLICIES.get(channel) or {}).items() if k not in ROUTING_KEYS} 

26 retval = own or config.DEFAULT_POLICY 

27 return retval

42def cwd_for(policy): 

43 # Expand ~ and $VARS at use-time (#54): a policy cwd like "~/g/git.voitta"

78def _gh_repos(tokens): 

79 """(repos, unresolvable) for a `gh` command. 

80 

81 **Every** target named, not the first one found. A command can name two -- 

82 `gh api repos/other/secret/issues -R mine/repo` -- and returning either one 

83 alone lets the other through: whichever is checked, the command still 

84 reaches the one that was not. So they are all collected and the caller 

85 requires all of them to be in scope. 

86 

87 `unresolvable` is True when a target is named in a way this cannot read: a 

88 path with a `..` segment in it, where the repo the text shows is not 

89 necessarily the repo the request reaches.""" 

90 repos = [] 

91 unresolvable = False 

92 for i, t in enumerate(tokens): 

93 # `GH_REPO=o/r gh issue list` -- an environment prefix retargets the 

94 # whole command, and shlex hands it to us as an ordinary token. 

95 if t.startswith("GH_REPO="): 

96 repos.append(t.split("=", 1)[1]) 

97 elif t in ("-R", "--repo") and i + 1 < len(tokens): 

98 repos.append(tokens[i + 1]) 

99 elif t.startswith("--repo="): 

100 repos.append(t.split("=", 1)[1]) 

101 elif t.startswith("-R") and len(t) > 2: 

102 repos.append(t[2:]) 

103 # `gh api repos/o/r/...` reaches any repo the token reaches, which is the 

104 # whole of this issue: read or write, and never through -R. Not gated on 

105 # the literal `api` token -- a user-defined `gh alias` expands to it inside 

106 # gh, so the word need never appear here. We are already inside a `gh` 

107 # command, so a token spelling `repos/OWNER/REPO` is an API path rather 

108 # than a coincidence. 

109 for t in tokens: 

110 m = _GH_API_PATH.match(t) 

111 if m: 

112 # `repos/mine/repo/../../other/secret` reads as `mine/repo` and is 

113 # not it. Rather than normalise -- and have to be sure the 

114 # normalisation matches whatever the API does -- refuse the shape. 

115 if ".." in t: 

116 unresolvable = True 

117 else: 

118 repos.append(m.group(1)) 

119 if not repos: 

120 # positional owner/repo, e.g. `gh repo view owner/repo` 

121 for t in tokens[1:]: 

122 if re.match(r"^[\w.-]+/[\w.-]+$", t): 

123 repos.append(t) 

124 break 

125 return (repos, unresolvable)

131def _git_dirs(tokens, base): 

132 """Every directory a git/gh command in this line would actually run in. 

133 

134 Not every directory the line *visits*: `pushd vendor && popd && git push` 

135 runs git at home, and recording `vendor` as a target refuses work that 

136 never touched it. So the line is walked as segments, the shell's directory 

137 is tracked across them, and a directory is recorded only where a git or gh 

138 command sits. 

139 

140 Four spellings retarget such a command and all of them were reachable past 

141 an earlier version of this check (#150, #186): `cd`/`pushd` move the shell, 

142 while `-C`, `--git-dir`/`--work-tree` and their `GIT_DIR=`/`GIT_WORK_TREE=` 

143 environment forms retarget one command without moving anything. git chains 

144 its own `-C` options -- `git -C a -C b` is `a/b` -- and only where the 

145 chain ends is a target.""" 

146 def _resolve(d, at): 

147 d = os.path.expanduser(d) 

148 p = os.path.normpath(d if os.path.isabs(d) else os.path.join(at, d)) 

149 # A gitdir names its worktree's parent; `git -C` wants the worktree. 

150 if os.path.basename(p) == ".git": 

151 p = os.path.dirname(p) 

152 return p 

153 

154 segments = [[]] 

155 for raw in tokens: 

156 if raw in _SEGMENT_BREAK: 

157 segments.append([]) 

158 continue 

159 segments[-1].append(raw) 

160 

161 out = [] 

162 cur = base 

163 last = None 

164 stack = [] 

165 subshell = [] 

166 for raw_seg in segments: 

167 if not raw_seg: 

168 continue 

169 # A subshell runs in its own directory and gives it back: after 

170 # `(cd vendor && ls)` the parent shell has not moved. Counting the 

171 # parentheses is what keeps `(cd x; ls); git push` and 

172 # `(cd x && ls) && git push` answering the same -- they used to differ, 

173 # which is worse than either answer alone. 

174 opens = len(raw_seg[0]) - len(raw_seg[0].lstrip("(")) 

175 closes = len(raw_seg[-1]) - len(raw_seg[-1].rstrip(")")) 

176 for _ in range(opens): 

177 subshell.append(cur) 

178 seg = [t.lstrip("({;&|").rstrip(")") for t in raw_seg] 

179 verb = seg[0] 

180 def _close(): 

181 for _ in range(closes): 

182 if subshell: 

183 return subshell.pop() 

184 return None 

185 

186 if verb == "popd": 

187 if stack: 

188 cur = stack.pop() 

189 _back = _close() 

190 if _back is not None: 

191 cur = _back 

192 continue 

193 if verb in ("cd", "pushd"): 

194 if verb == "pushd": 

195 stack.append(cur) 

196 prev = cur 

197 # `cd` with no argument, or `cd ~`, is home -- the same rule the 

198 # grant layer's own cd tracking uses. `cd -` is the last place the 

199 # shell was. 

200 if len(seg) == 1 or seg[1] == "~": 

201 cur = os.path.expanduser("~") 

202 elif seg[1] == "-": 

203 cur = last if last is not None else cur 

204 else: 

205 cur = _resolve(seg[1], cur) 

206 last = prev 

207 _back = _close() 

208 if _back is not None: 

209 cur = _back 

210 continue 

211 target = None 

212 i = 0 

213 while i < len(seg): 

214 t = seg[i] 

215 if t == "-C" and i + 1 < len(seg): 

216 target = _resolve(seg[i + 1], target or cur) 

217 i += 2 

218 continue 

219 if t.startswith("-C") and len(t) > 2: 

220 target = _resolve(t[2:], target or cur) 

221 for flag in ("--git-dir", "--work-tree"): 

222 if t == flag and i + 1 < len(seg): 

223 target = _resolve(seg[i + 1], cur) 

224 elif t.startswith(flag + "="): 

225 target = _resolve(t.split("=", 1)[1], cur) 

226 for var in ("GIT_DIR=", "GIT_WORK_TREE="): 

227 if t.startswith(var): 

228 target = _resolve(t.split("=", 1)[1], cur) 

229 i += 1 

230 if any(os.path.basename(t) in ("git", "gh") for t in seg): 

231 out.append(target or cur) 

232 _back = _close() 

233 if _back is not None: 

234 cur = _back 

235 return out

260# What ends a command for the segment scan: the list and pipe separators, a 

261# newline, and the edges of a subshell, a `$( )` or a backquote, so that 

262# `echo ok;gh ...` and `echo $(gh ...)` each give gh a segment of its own. 

263_SEG_PUNCT = ";&|\n()`"

266def _shell_tokens(command): 

267 """shlex tokens with the separators split out even where they touch a word 

268 (`ok;git`, `x&&cd d`), each run as its own token, a newline read as `;`. 

269 `_git_dirs` walks segments by those tokens, and `shlex.split` leaves them 

270 glued, which hid a `-C` or `cd` from it. None if the text will not lex.""" 

271 try: 

272 lex = shlex.shlex(command, posix=True, punctuation_chars=";&|\n") 

273 lex.whitespace = " \t\r" 

274 lex.whitespace_split = True 

275 raw = list(lex) 

276 except ValueError: 

277 return None 

278 retval = [] 

279 for t in raw: 

280 if t and set(t) <= set(";&|\n"): 

281 t = t if t in _SEGMENT_BREAK else ";" 

282 retval.append(t) 

283 return retval

286def _mentions(command, name): 

287 """True when `name` appears as a word anywhere in the raw text, glued to a 

288 separator or not. Over-reports on purpose (`grep gh f` counts): it only 

289 decides whether the repo checks run, and shlex keeps `ok;gh` one word.""" 

290 retval = re.search(rf"(?<![\w.-]){re.escape(name)}(?![\w.-])", command) is not None 

291 return retval

294def _gh_segment_tokens(command, name="gh"): 

295 """The segments of `command` that invoke gh, each a token list, or None 

296 when the command cannot be split into segments. Kept apart, because each 

297 segment's positional repo is its own target (#425 review): concatenated, 

298 `gh repo view a/b; gh repo view c/d` read as naming only `a/b`. 

299 

300 The repo scans in `_gh_repos` read a `-R` value and a positional 

301 `owner/repo`. Run over the whole command, they read other commands' 

302 arguments as gh's: `jq . app/data.json` beside a `gh run list` was refused 

303 as repo 'app/data.json', and `grep -R pat dir; gh pr list` would read `pat` 

304 as one. A newline separates commands as `;` does, which `shlex.split` alone 

305 treats as plain whitespace.""" 

306 try: 

307 lex = shlex.shlex(command, posix=True, punctuation_chars=_SEG_PUNCT) 

308 lex.whitespace = " \t\r" 

309 lex.whitespace_split = True 

310 tokens = list(lex) 

311 except ValueError: 

312 return None 

313 out, seg = [], [] 

314 for t in tokens + [";"]: 

315 if t and set(t) <= set(_SEG_PUNCT): 

316 if _invokes(seg, name): 

317 out.append(seg) 

318 seg = [] 

319 else: 

320 seg.append(t) 

321 retval = out 

322 return retval

325def _check_github(command, policy): 

326 allowed = policy.get("github_repos") or [] 

327 if not allowed: 

328 return (True, "") 

329 tokens = _tokens(command) 

330 # By basename: `/usr/bin/git push` and `./gh api ...` are the same commands 

331 # as `git` and `gh`, and an exact token match answered "no git here" to 

332 # both, skipping the whitelist entirely. Pre-dates #150 and #186; found by 

333 # the third adversarial pass on #195. 

334 is_gh = _invokes(tokens, "gh") or _mentions(command, "gh") 

335 is_git = _invokes(tokens, "git") or _mentions(command, "git") 

336 if not (is_gh or is_git): 

337 return (True, "") 

338 # A git command that names a GitHub URL outright (`git ls-remote 

339 # git@github.com:o/r`, `git push https://github.com/o/r`) targets that 

340 # repo, not the checkout's origin -- and since #122 every channel's git 

341 # carries the operator's credential over https, so the named repo is what 

342 # the whitelist has to be checked against. 

343 def _in_scope(repo): 

344 # GitHub owner and repo names are case-insensitive, so `mine/*` has to 

345 # admit `MINE/Repo`; the comparison was case-sensitive and refused it. 

346 r = repo.lower() 

347 return any(fnmatch.fnmatch(r, pat.lower()) for pat in allowed) 

348 

349 named = [m.group(1) for m in (_GITHUB_URL.match(t) for t in tokens) if m] if is_git else [] 

350 for repo in named: 

351 if not _in_scope(repo): 

352 return (False, f"repo '{repo}' not in channel whitelist {allowed}") 

353 # A git segment that names no GitHub URL reaches its checkout's origin. 

354 # Decided per segment (#425 review): an allowed URL in one segment says 

355 # nothing about `git status` in the next. 

356 git_segments = _gh_segment_tokens(command, "git") if is_git else [] 

357 if git_segments is None: 

358 git_segments = [tokens] 

359 git_bare = any(not any(_GITHUB_URL.match(t) for t in seg) for seg in git_segments) 

360 gh_segments = _gh_segment_tokens(command) if is_gh else None 

361 if is_gh and gh_segments is None: 

362 gh_segments = [tokens] 

363 gh_repos, gh_unresolvable, gh_tokens = [], False, [] 

364 # A gh segment naming no repo reaches the checkout's origin, which then 

365 # has to be in scope too, whatever the other segments name. 

366 gh_bare = False 

367 gh_api_bare = False 

368 if is_gh: 

369 # `export GH_REPO=o/r; gh pr list` may retarget gh from another 

370 # segment, so such a value is one more target to check -- never the 

371 # segment's own: `echo GH_REPO=o/r; gh issue list` sets nothing, and 

372 # counting it would excuse that bare gh from the origin check below. 

373 exported = [t.split("=", 1)[1] for t in tokens if t.startswith("GH_REPO=")] 

374 gh_repos += exported 

375 for seg in gh_segments: 

376 seg_repos, seg_bad = _gh_repos(seg) 

377 gh_repos += seg_repos 

378 gh_unresolvable = gh_unresolvable or seg_bad 

379 gh_bare = gh_bare or not seg_repos 

380 # Per segment, like the targets: an allowed repo named elsewhere 

381 # says nothing about where this segment's `gh api` goes. 

382 gh_api_bare = gh_api_bare or ("api" in seg and not seg_repos) 

383 gh_tokens += seg 

384 if gh_unresolvable: 

385 return (False, "gh names a repo path this policy cannot resolve; blocked by channel policy") 

386 # `gh api` that names no repo -- `gh api graphql`, `gh api user` -- is 

387 # refused rather than waved through when a whitelist exists. The target may 

388 # be inside a GraphQL document or absent entirely, and "I could not tell" 

389 # has to mean no while `github_repos` is set, or the one command that 

390 # reaches every repo is the one command that is never checked. 

391 if is_gh and gh_api_bare: 

392 return (False, "gh api names no repo this policy can resolve; blocked by channel policy") 

393 # Every named target, not the first: a command naming two reaches both. 

394 for repo in gh_repos: 

395 if not _in_scope(repo): 

396 return (False, f"repo '{repo}' not in channel whitelist {allowed}") 

397 # Every explicit target is checked above; none of them excuses a segment 

398 # that named nothing (#425 review). Only when no git or gh segment is 

399 # bare is the command done. 

400 if not (git_bare or (is_gh and gh_bare)): 

401 return (True, "") 

402 # Every directory the command names, not just the channel's cwd: a `cd` 

403 # and a `-C` retarget git identically (#186), and gh too reads the 

404 # checkout it runs in. Each one's origin has to be in scope, because the 

405 # command reaches all of them. 

406 _base = cwd_for(policy) 

407 _dirs = _git_dirs(_shell_tokens(command) or tokens, _base) 

408 for _d in _dirs: 

409 _r = _git_origin(_d) 

410 if _r and not _in_scope(_r): 

411 return (False, f"repo '{_r}' not in channel whitelist {allowed}") 

412 repo = _git_origin(_dirs[-1] if _dirs else _base) 

413 if not repo: 

414 return (False, "target github repo undeterminable; blocked by channel policy") 

415 if _in_scope(repo): 

416 return (True, "") 

417 return (False, f"repo '{repo}' not in channel whitelist {allowed}")

438_EGRESS_VERBS = ("curl", "wget")

440# Flags that turn a fetch into an upload (#222). #149 wrote down its own premise 

441# -- "a `curl -X POST` is already mutating and already parks" -- and on 

442# voitta-yolt 1.6.0 that premise is false in the worst direction: curl's rule 

443# default was `safe` there, so `curl -d@/etc/passwd https://<allow-listed host>` 

444# came back `safe`, and `safe` short-circuits to execute() without the grant 

445# layer ever being consulted. Not "grant vouched for it": grant never saw it. 

446# 2.0.0 moved the default to `ask`, so the exposure is exactly a deployment 

447# running a classifier older than that -- which the version floor does not 

448# prevent, because preflight only warns. 

449# 

450# So this stops depending on the classifier having caught them. A request body 

451# is an upload; "read-only" in #149's sense is about this machine, and the host 

452# being allow-listed says where bytes may go, not that bytes may go. 

453_BODY_FLAGS_LONG = ( 

454 "--data", "--data-raw", "--data-binary", "--data-urlencode", "--data-ascii", 

455 "--json", "--form", "--form-string", "--upload-file", "--request", 

456 # wget's spellings 

457 "--post-data", "--post-file", "--body-data", "--body-file", "--method", 

458) 

459# ...and the short forms, which is where the care goes. In a short-option 

460# cluster every character is an option letter, so `-sXPOST` and `-d@f` are the 

461# same option as `-X POST` and `-d @f` -- that is the bug voitta-yolt#156 fixed 

462# upstream, and repeating it here would reopen this hole one spelling over. 

463# Case matters: `-d` is data but `-D` dumps headers, `-F` is form but `-f` is 

464# fail, `-T` uploads but `-t` does not, `-X` sets the method but `-x` is a proxy. 

465# Ordinary curl is unaffected -- `-sSfL`, `-fsSL`, `-I`, `-o`, `-v` contain none 

466# of these four letters. 

467# 

468# Per verb, because the same letters mean different things. wget spells every 

469# body option long (`--post-data`, `--body-file`), and its short flags collide 

470# head-on: `-d` is debug, `-T` is a timeout, `-F` is --force-html. Carrying 

471# curl's letters across would card ordinary wget use for nothing. 

472_BODY_LETTERS = {"curl": frozenset("dFTX"), "wget": frozenset()} 

473 

474# Options that source further options or URLs from a file, which this layer 

475# cannot read -- so what the command does is not what the command says. 

476# Measured: a config file containing `data = "@/tmp/x"` makes curl POST that 

477# file, and curl itself prints "POST is already inferred" while the argv scan 

478# sees only `-K`. Same for a destination: an `output` line took effect from a 

479# file that never appeared in argv. 

480# 

481# `-i` is the reason this is per-verb too: wget's reads a URL list, curl's is 

482# --include and is everywhere. Getting that backwards would card every 

483# `curl -i` on the box. 

484_INDIRECT_LONG = ("--config", "--input-file") 

485_INDIRECT_LETTERS = {"curl": frozenset("K"), "wget": frozenset("i")}

493def _upload_reason(verb, tokens): 

494 """Why this fetch is not read-only, or None. 

495 

496 Refuses on the letter rather than pairing it with its value. Deciding which 

497 letter in a cluster consumes what is precisely the reasoning that produced 

498 the upstream matcher bug, and the cost of not doing it is one approval card 

499 for `curl -X GET`, an unusual spelling of a thing that needs no body.""" 

500 body = _BODY_LETTERS.get(verb, frozenset()) 

501 indirect = _INDIRECT_LETTERS.get(verb, frozenset()) 

502 retval = None 

503 for t in tokens: 

504 if _long_hit(t, _BODY_FLAGS_LONG): 

505 retval = "fetch carries a request body (upload), so it is not read-only" 

506 break 

507 if _long_hit(t, _INDIRECT_LONG): 

508 retval = f"fetch takes options or URLs from a file ({t}), so what it does is not visible here" 

509 break 

510 if t.startswith("-") and not t.startswith("--"): 

511 letters = set(t[1:]) 

512 if letters & body: 

513 retval = "fetch carries a request body (upload), so it is not read-only" 

514 break 

515 if letters & indirect: 

516 retval = f"fetch takes options or URLs from a file ({t}), so what it does is not visible here" 

517 break 

518 return retval

520# git talks to a remote over https without any of the above (#149 review). Only 

521# these subcommands do: `git log --grep https://x` names a URL and contacts 

522# nothing, and carding it would be a guard inventing work. 

523_GIT_NET_SUBS = ("clone", "fetch", "pull", "push", "ls-remote", "submodule", "remote")

526def _host_of(authority): 

527 """The host inside a URL authority: userinfo dropped, port dropped, IPv6 

528 literal kept whole. `https://evil.test@example.com/` contacts example.com, 

529 which is why userinfo goes before the port split rather than after.""" 

530 hostport = authority.rsplit("@", 1)[-1] 

531 if hostport.startswith("["): # [2001:db8::1]:443 -- the colons are the address 

532 retval = hostport[: hostport.index("]") + 1] if "]" in hostport else hostport 

533 return retval.lower() 

534 retval = hostport.split(":", 1)[0].lower() 

535 return retval

538def _git_subcommand(tokens): 

539 """The subcommand in a `git` invocation, skipping git's own flags and the 

540 values of the two that take one.""" 

541 retval = None 

542 i = len(tokens) 

543 for n, t in enumerate(tokens): 

544 if os.path.basename(t) == "git": 

545 i = n + 1 

546 break 

547 while i < len(tokens): 

548 t = tokens[i] 

549 if t in ("-C", "-c", "--git-dir", "--work-tree", "--namespace"): 

550 i += 2 

551 continue 

552 if t.startswith("-"): 

553 i += 1 

554 continue 

555 retval = t 

556 break 

557 return retval

560def check_egress(command, policy): 

561 """(ok, reason) for the network reach of a read-only command (#149). 

562 

563 `curl` and `wget` are read-only to YOLT, so they auto-run with no card, to 

564 any host, and so does `git ls-remote https://<anywhere>`. Everything else that leaves the box -- `nc`, `ssh`, `scp`, a 

565 `curl -X POST` -- is already mutating and already parks. That left one 

566 uncarded path off the machine, and the sandbox cannot help: it confines the 

567 filesystem, not the network. What is left to send is whatever the channel 

568 may legitimately read, which is its own tree -- a project `.env` or 

569 `terraform.tfvars` is one `cat` and one `curl` away. 

570 

571 So a fetch is allowed without a card only when every host it names is in 

572 the channel's `allow_domains`. Anything else is *mutating*, not blocked: it 

573 parks for a trusted user, who can say yes. A channel with no 

574 `allow_domains` cards every fetch, which is the honest default -- the 

575 alternative is a built-in list that is wrong for somebody. 

576 

577 The same applies to the git subcommands that contact a remote; the ones 

578 that do not (`git log --grep https://x`) are left alone. 

579 

580 A host has to be statically visible in the command, which means a scheme. 

581 `curl example.com` and `curl "$URL"` park rather than being guessed at: 

582 this is a textual guard like `exclude`, and it says so instead of pretending 

583 to parse a shell.""" 

584 tokens = _tokens(command) 

585 fetches = any(os.path.basename(t) in _EGRESS_VERBS for t in tokens) 

586 if not fetches and any(os.path.basename(t) == "git" for t in tokens): 

587 fetches = _git_subcommand(tokens) in _GIT_NET_SUBS 

588 if not fetches: 

589 return (True, "") 

590 # Before the host check, and deliberately not subject to it: an allow-listed 

591 # host says where bytes may go, not that bytes may go (#222). Scanned from 

592 # the first fetch verb so a body-shaped flag belonging to an earlier command 

593 # in a chain -- `grep -d skip x && curl https://ok` -- does not card the 

594 # fetch. Over-matching past that point is left alone: it costs a card, which 

595 # is the direction this is allowed to be wrong in. 

596 _first = next((i for i, t in enumerate(tokens) 

597 if os.path.basename(t) in _EGRESS_VERBS), None) 

598 if _first is not None: 

599 _why = _upload_reason(os.path.basename(tokens[_first]), tokens[_first + 1:]) 

600 if _why: 

601 return (False, _why) 

602 hosts = [_host_of(h) for h in _URL_HOST.findall(command)] 

603 if not hosts: 

604 return (False, "fetch: no statically known host (use an explicit https:// URL)") 

605 for host in hosts: 

606 if not host_allowed(host, policy): 

607 return (False, f"fetch to '{host}' is not in this channel's allow_domains") 

608 return (True, "")

611def host_of_url(url): 

612 """The host inside a URL, or None. Public because the grant layer resolves 

613 a git remote's configured URL when the command itself names no host 

614 (#253) -- one parser for "which host is this", not two.""" 

615 m = _URL_HOST.search(url or "") 

616 if m: 

617 retval = _host_of(m.group(1)) 

618 return retval 

619 # scp-style, which git accepts for ssh remotes: user@host:path 

620 m = re.match(r"^[^/\s]*@([^:/\s]+):", url or "") 

621 retval = _host_of(m.group(1)) if m else None 

622 return retval

625def host_allowed(host, policy): 

626 """Is this host in the channel's `allow_domains` (#149)? 

627 

628 One rule, two callers: the textual guard over a shell command, and the 

629 web_fetch tool (#62). A second copy of the matching would be a second 

630 answer to "may this channel reach that host", and the two would drift.""" 

631 allowed = [p.lower() for p in (policy.get("allow_domains") or [])] 

632 retval = any(fnmatch.fnmatch((host or "").lower(), pat) for pat in allowed) 

633 return retval

636def _norm_path(p, base): 

637 p = os.path.expanduser(os.path.expandvars(p)) 

638 if not os.path.isabs(p): 

639 p = os.path.join(base, p) 

640 retval = os.path.normpath(p) 

641 return retval

644def _check_exclude(command, policy): 

645 """Textual guard against a command touching an excluded subtree (#55). 

646 

647 Not the containment -- that is the sandbox (#116, sandbox.py), which denies 

648 the same paths in the kernel and so also catches a symlink or a path the 

649 shell resolves at runtime. This runs first to block the obvious textual 

650 cases (`cat ~/g/OneDrive/x`, `cd <excluded>`) with a reason the agent can 

651 read, instead of a bare "Operation not permitted" from the shell.""" 

652 excludes = policy.get("exclude") or [] 

653 if not excludes: 

654 return (True, "") 

655 base = cwd_for(policy) 

656 ex_norm = [_norm_path(p, base) for p in excludes] 

657 for tok in _tokens(command): 

658 if "/" not in tok and "~" not in tok: 

659 continue # not path-shaped; skip 

660 cand = _norm_path(tok, base) 

661 for ex in ex_norm: 

662 if cand == ex or cand.startswith(ex + os.sep): 

663 return (False, f"'{tok}' resolves under excluded path {ex}") 

664 return (True, "")

667def _check_self(command, policy): 

668 """Block a command that names this deployment's own config or policy file 

669 (#147). 

670 

671 Writing one is a capability change -- the policy file is `cwd`, 

672 `allow_read`/`allow_write`, `exclude`, `env` and `env_passthrough` -- and 

673 the grant layer would run it as an ordinary in-tree write, with no card, 

674 whenever a channel's cwd is the directory shmobster was started from. 

675 

676 This is the readable reason, not the defence: it matches tokens, so a 

677 shell variable, a glob or an `sh -c` hides the path from it. The sandbox 

678 denies both operations on both files in the kernel (sandbox.py), which is 

679 what actually holds. This runs first so the agent is told why instead of 

680 reading "Operation not permitted" off a shell. 

681 

682 Reads are refused as well as writes, and deliberately: the values are 

683 `${VAR}` references rather than literals (#73), but a config value is not 

684 something this agent posts into a channel under any circumstances. 

685 Changing a channel's policy has a route already -- set_policy, trusted 

686 users only.""" 

687 if not config.SELF_FILES: 

688 return (True, "") 

689 base = cwd_for(policy) 

690 for tok in _tokens(command): 

691 if "/" not in tok and "." not in tok: 

692 continue # neither path-shaped nor a bare filename 

693 cand = os.path.realpath(_norm_path(tok, base)) 

694 if cand in config.SELF_FILES: 

695 return (False, f"'{tok}' is this deployment's own config; use set_policy to change a policy") 

696 return (True, "")

699# Which argument a writing verb actually writes (#174 review). "A spine path 

700# appears next to a write verb" is not the question: `cp workspace/SOUL.md 

701# backup.md` reads the spine and writes somewhere else, and `sed -n 1,5p 

702# workspace/SOUL.md` writes nothing at all. Getting this wrong blocks ordinary 

703# work in the deployment's own tree, which is a guard nobody keeps. 

704# 

705# Verb list mirrors grant.FS_VERBS by hand rather than importing it: grant 

706# imports this module, so the arrow goes one way. A verb missing here costs a 

707# readable message, not the denial -- the sandbox is what holds. 

708def _write_targets(tokens): 

709 """The paths this command writes, as written. Best effort by design.""" 

710 targets = set() 

711 for i, tok in enumerate(tokens): 

712 if tok in (">", ">>") and i + 1 < len(tokens): 

713 targets.add(tokens[i + 1]) 

714 if not tokens: 

715 return targets 

716 verb = os.path.basename(tokens[0]) 

717 args = [t for t in tokens[1:] if not t.startswith("-")] 

718 if verb in ("cp", "mv", "ln", "install") and len(args) >= 2: 

719 targets.add(args[-1]) # ... SOURCE DEST 

720 elif verb in ("tee", "touch"): 

721 targets.update(args) # every file argument is written 

722 elif verb == "chmod": 

723 targets.update(args[1:]) # the mode is not a path 

724 elif verb == "sed" and any(t == "-i" or t.startswith("-i") for t in tokens): 

725 targets.update(args[1:]) # in place only; the script is not a path 

726 elif verb == "dd": 

727 targets.update(t.split("=", 1)[1] for t in tokens if t.startswith("of=")) 

728 retval = targets 

729 return retval

732def _check_spine(command, policy): 

733 """Block a command that writes the agent's own standing prompt (#174). 

734 

735 `spine.load_system_prompt()` reads these files into the system prompt every 

736 turn, and when the bundled `./workspace` sits inside a channel's tree the 

737 grant layer runs `tee workspace/SOUL.md` as an ordinary in-tree write, with 

738 no card. That is not widening the envelope, it is editing the instructions 

739 that say how to behave inside it -- including the ones about being honest 

740 about what has been read (#134). #130 already refuses a channel `skills` 

741 entry under a writable root for the same reason; this is that hazard with a 

742 shorter path. 

743 

744 Writes only, and only where the spine is the *target*. The spine is the 

745 agent's persona, not a secret, and a channel reads and copies its own tree 

746 legitimately: `cat workspace/SOUL.md`, `grep terse workspace/SOUL.md > 

747 /tmp/out` and `cp workspace/SOUL.md backup.md` all pass, because none of 

748 them writes the spine.""" 

749 paths = spine.files() 

750 if not paths: 

751 return (True, "") 

752 base = cwd_for(policy) 

753 for tok in _write_targets(_tokens(command)): 

754 if os.path.realpath(_norm_path(tok, base)) in paths: 

755 return (False, f"'{tok}' is this agent's own standing prompt; it changes " 

756 f"by a human edit or a PR, not from a channel") 

757 return (True, "")

764 return (False, reason)
