1"""The channel repo's own instructions, as standing context (#259). 

2 

3A repo usually documents how it wants to be worked on, and the agent standing 

4in that repo never read it. The case that produced this: `scrooge-banking` has 

5`docs/BRANCHING-STRATEGY.md` and `docs/PERSONAL_BRANCHES.md`, both committed, 

6both saying that a collaborator's branch is what deploys to their subdomain. 

7Six weeks of UX work sat on a feature branch instead, invisible, and when asked 

8why, the agent read `git log` for one branch, found nothing, and reported the 

9work had never been done. The answer was a file in the tree it was standing in. 

10 

11That is the agent's job rather than the collaborator's, and it is the reason 

12this deployment exists: the architect sets up the workflow, and the people in 

13the channel are specialists -- a UX designer describing a screen should not 

14have to know which branch makes it appear. 

15 

16**Read from a commit, never from the working tree.** This is the whole safety 

17argument, and it is why `memory.py`'s rule could not simply be reused. Memory 

18refuses any file under a writable root, because the grant layer runs an in-tree 

19write with no card and a writable prompt file lets one `cat > FILE` become the 

20next turn's instructions. The file wanted here is INSIDE the channel's tree by 

21definition. So the content comes from `git show <rev>:<path>` -- an uncommitted 

22edit, which is what a turn can make silently, steers nothing. 

23 

24A turn can still commit, and in an unattended channel (#253) it can do so with 

25no card. That is deliberate and bounded: a commit is a reviewable object with 

26an author and a diff, in the channel's own repo, which is the scope its 

27operator already drew. The distinction being preserved is between an edit that 

28leaves a trace and one that does not. 

29""" 

30import logging 

31import os 

32import posixpath 

33import subprocess 

34 

35 

36# Conventional names first, then whatever the channel names for itself. The 

37# conventions are the ones agent tooling already writes: a repo that has one 

38# expects an agent to read it. 

39DEFAULT_PATHS = ("CLAUDE.md", "AGENTS.md", "docs/CLAUDE.md", "docs/AGENTS.md") 

40 

41# A budget, not an archive. A repo's docs directory can be a book; the system 

42# prompt cannot. Truncation says where it stopped rather than trailing off. 

43_MAX_FILE = 6000 

44_MAX_TOTAL = 16000 

45_REV = "HEAD" 

46 

47 

48def _git(directory, *args): 

49 """stdout bytes of a git command in `directory`, or None.""" 

50 try: 

51 proc = subprocess.run( 

52 ["git", "-C", directory, *args], 

53 stdout=subprocess.PIPE, stderr=subprocess.DEVNULL, 

54 timeout=10, 

55 ) 

56 except (OSError, subprocess.SubprocessError): 

57 return None 

58 retval = proc.stdout if proc.returncode == 0 else None 

59 return retval

62def _safe(path): 

63 retval = bool(path) and not path.startswith("/") and ".." not in path.split("/") 

64 return retval 

65

67def _entry(directory, path): 

68 """(mode, type, sha) of `path` in the HEAD tree, or None. 

69 

70 `--full-tree`, because `git show HEAD:<path>` reads from the repository 

71 root and a channel's cwd can be a subdirectory of its repo; without it the 

72 two calls would look at different files. And only an entry whose path IS 

73 the one asked for: `ls-tree` on a directory lists its children, and the 

74 first child must not stand in for the directory.""" 

75 out = _git(directory, "ls-tree", "-z", "--full-tree", _REV, "--", path) if _safe(path) else None 

76 head, _, name = (out or b"").split(b"\0", 1)[0].partition(b"\t") 

77 fields = head.decode("ascii", "replace").split(" ") 

78 retval = tuple(fields) if len(fields) == 3 and name == path.encode() else None 

79 return retval

82def _resolve(directory, path): 

83 """The in-repo path whose blob holds `path`'s text at HEAD, or None (#379). 

84 

85 A committed symlink is a blob whose content is the link TEXT, so 

86 `git show HEAD:CLAUDE.md` on `CLAUDE.md -> AGENTS.md` returns the nine 

87 characters "AGENTS.md". Repos that keep one instruction file under both 

88 names do exactly this. Follow one level, to a relative target inside the 

89 repo, by the same rule a named path has to pass; a link to a link, or out 

90 of the tree, is not followed.""" 

91 entry = _entry(directory, path) 

92 if not entry or entry[1] != "blob": 

93 return None 

94 if entry[0] != "120000": 

95 return path 

96 raw = _git(directory, "cat-file", "blob", entry[2]) 

97 target = raw.decode("utf-8", "replace") if raw else "" 

98 if not _safe(target): 

99 return None 

100 target = posixpath.normpath(posixpath.join(posixpath.dirname(path), target)) 

101 entry = _entry(directory, target) 

102 retval = target if entry and entry[1] == "blob" and entry[0] != "120000" else None 

103 return retval

106def _read(directory, path): 

107 """One file's contents at HEAD, or None. 

108 

109 `--` separates the pathspec from revisions so a file named like a ref 

110 cannot be read as one, and a path outside the repo simply fails: `git show` 

111 resolves against the repository root, not the filesystem.""" 

112 out = _git(directory, "show", f"{_REV}:{path}") if _safe(path) else None 

113 if out is None: 

114 return None 

115 # Bytes, decoded here rather than by subprocess: git will hand back 

116 # whatever is in the blob, and a document that is not valid UTF-8 would 

117 # raise UnicodeDecodeError on EVERY turn -- this runs before the model 

118 # call, so one bad commit would take the channel down until someone 

119 # changed it (Codex adversarial review, #261). Replacement characters in a 

120 # prompt are survivable; a channel that cannot answer is not. 

121 retval = out.decode("utf-8", "replace") 

122 return retval

125def files(policy): 

126 """[(path, text)] for this channel's project docs, in order. 

127 

128 The defaults plus whatever `project_docs` names: `docs/BRANCHING-STRATEGY.md` 

129 is the file that mattered here and no convention would have guessed it.""" 

130 # An explicit cwd, never `cwd_for`'s fallback. Without this a channel that 

131 # names no directory reads the DEPLOYMENT's own repo -- shmobster's 

132 # CLAUDE.md, which is written for the people who develop the agent -- into 

133 # a Slack channel's prompt. Caught by the self-check, and it would have 

134 # been invisible in production because the block reads plausibly. 

135 directory = policy.get("cwd") 

136 directory = os.path.expanduser(os.path.expandvars(directory)) if directory else None 

137 if not directory or not os.path.isdir(directory): 

138 return [] 

139 # Named files first, conventions second. `project_docs` is an operator 

140 # saying "this one matters"; the defaults are a guess that a file with a 

141 # conventional name is worth reading. Spending the budget on the guess and 

142 # starving the instruction the channel exists to follow is the wrong way 

143 # round -- and it is what happened here before review (#261). 

144 named = [p for p in (policy.get("project_docs") or [])] 

145 wanted = named + [p for p in DEFAULT_PATHS if p not in named] 

146 retval, total, omitted, seen = [], 0, [], set() 

147 for path in wanted: 

148 # Through a symlink to the file it names, and each file once: a repo 

149 # with CLAUDE.md -> AGENTS.md would otherwise spend the budget twice. 

150 path = _resolve(directory, path) 

151 if not path or path in seen: 

152 continue 

153 seen.add(path) 

154 body = _read(directory, path) 

155 if not body or not body.strip(): 

156 continue 

157 if len(body) > _MAX_FILE: 

158 body = body[:_MAX_FILE] + f"\n...[truncated at {_MAX_FILE} characters]" 

159 if total + len(body) > _MAX_TOTAL: 

160 # Keep going rather than stopping: one oversized file must not hide 

161 # every smaller one after it. What is dropped is named in the 

162 # prompt, so the reader knows the set is incomplete. 

163 logging.info("projectdocs: %s omitted, budget spent", path) 

164 omitted.append(path) 

165 continue 

166 total += len(body) 

167 retval.append((path, body)) 

168 if omitted: 

169 retval.append(("(omitted)", "These were not included, the budget was " 

170 "spent: " + ", ".join(omitted) + ". Read them with a " 

171 "command if the answer might be in one.")) 

172 return retval

175def _fence(body): 

176 """A fence longer than any run of backticks inside, so a doc containing a 

177 code block cannot end the block it is quoted in.""" 

178 longest = 0 

179 run = 0 

180 for ch in body: 

181 run = run + 1 if ch == "`" else 0 

182 longest = max(longest, run) 

183 retval = "`" * max(3, longest + 1) 

184 return retval

187def prompt_block(policy): 

188 """The project-instructions block, or "" when the repo carries none.""" 

189 found = files(policy or {}) 

190 if not found: 

191 retval = "" 

192 return retval 

193 parts = [ 

194 "## How this project asks to be worked on\n", 

195 "These files are committed in this channel's own repository, read at " 

196 f"`{_REV}`. They are the project's instructions -- the working " 

197 "agreements the people here rely on, such as which branch deploys " 

198 "where -- and following them is part of doing the work correctly. Cite " 

199 "the file when you apply a rule from it, so anyone can check you.\n", 

200 "Read from the last commit, not the working copy, so an uncommitted " 

201 "edit cannot change what you are told mid-task.\n", 

202 ] 

203 for path, body in found: 

204 fence = _fence(body) 

205 parts.append(f"### {path}\n\n{fence}\n{body.rstrip()}\n{fence}\n") 

206 retval = "\n".join(parts) 

207 return retval
