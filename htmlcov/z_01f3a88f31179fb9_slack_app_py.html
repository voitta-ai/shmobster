33def _thread_context(client, channel, thread_ts, cur_ts): 

34 """Flatten prior thread messages into a labeled transcript (Iter 11 / #11). 

35 

36 Flattened (not native assistant turns) so multi-user threads with consecutive 

37 same-role messages don't trip vendor role-alternation rules. Long-term memory 

38 (workspace MEMORY.md) is a separate, later concern. 

39 """ 

40 if not thread_ts: 

41 return None 

42 try: 

43 resp = client.conversations_replies( 

44 channel=channel, ts=thread_ts, limit=_MAX_THREAD_MSGS 

45 ) 

46 except Exception: 

47 return None 

48 lines = [] 

49 for m in resp.get("messages", []): 

50 if m.get("ts") == cur_ts: 

51 continue # the current mention -- handler adds it as the user turn 

52 who = identity.speaker(m) # self / sibling agent / human, not blanket "agent" (#60) 

53 text = (m.get("text") or "").strip() 

54 if text: 

55 lines.append(f"[{who}] {text}") 

56 retval = "\n".join(lines) or None 

57 return retval

71def _remember_files(channel, thread_ts, files): 

72 """Record what a mention attached, for the turns that resume its task. 

73 

74 While the thread's task is under way (a request parked, running, or a 

75 resume in flight: approvals.busy), a mention adds its files to the task's 

76 and one without files changes nothing: a "how is it going?" is not a new 

77 task, and clearing then would starve the resume this exists for. Once the 

78 thread is idle, a mention with files starts the list afresh and one 

79 without clears it, so a later, unrelated task does not inherit an old 

80 file.""" 

81 key = (channel, thread_ts) 

82 busy = approvals.busy(channel, thread_ts) 

83 refs = [{k: f.get(k) for k in _FILE_REF_KEYS} for f in files or []] 

84 if refs: 

85 held = _TASK_FILES.pop(key, None) or [] 

86 if not busy: 

87 held = [] 

88 seen = {r.get("url_private_download") or r.get("url_private") for r in held} 

89 merged = held + [r for r in refs 

90 if (r.get("url_private_download") or r.get("url_private")) not in seen] 

91 _TASK_FILES[key] = merged[-_TASK_FILES_PER_THREAD:] # one resume reads them all 

92 if len(_TASK_FILES) > _TASK_FILES_MAX: 

93 del _TASK_FILES[next(iter(_TASK_FILES))] 

94 elif not busy: 

95 _TASK_FILES.pop(key, None)

104def _reply_payload(reply): 

105 """A model reply as a `markdown` block, with `text` as the notification 

106 fallback and the copy identity.agent_name reads from history. 

107 

108 The model writes standard Markdown; posted as bare `text`, Slack reads it as 

109 mrkdwn, so `**CRM:**` showed its asterisks and `[name](url)` stayed literal. 

110 A markdown block leaves the rendering to Slack. Past the block's cap the 

111 reply goes out as plain text, as before.""" 

112 retval = {"text": reply} 

113 if len(reply) <= _MARKDOWN_MAX: 

114 retval["blocks"] = [{"type": "markdown", "text": reply}] 

115 return retval

131def _post_full_command(client, channel, thread_ts, req_id, command): 

132 """Post a command too long for its card as a plain thread reply and return 

133 its permalink, or None. Message text has no 3000-character block cap, and 

134 this needs only `chat:write` -- a file upload would need `files:write`, 

135 which the manifest leaves out on purpose. Never raises: the card must post 

136 whether or not this did.""" 

137 cmd = redact.scrub(command) 

138 if len(cmd) > _FULL_COMMAND_MAX: 

139 cmd = cmd[:_FULL_COMMAND_MAX] + "\n... (truncated)" 

140 retval = None 

141 try: 

142 resp = client.chat_postMessage( 

143 channel=channel, thread_ts=thread_ts, 

144 text=f"Full command for [{req_id}]:\n```{cmd}```", 

145 ) 

146 retval = client.chat_getPermalink(channel=channel, message_ts=resp["ts"]).get("permalink") or None 

147 except Exception: 

148 logging.exception("could not post the full command for %s", req_id) 

149 return retval

152# Slack's own "try again" answers, which can arrive as ok:false at HTTP 200. 

153_TRANSIENT_ERRORS = {"internal_error", "fatal_error", "service_unavailable", "request_timeout", "ratelimited"} 

154# A "transient" failure that repeats this many times on one card is treated as 

155# a rejection: some payloads get internal_error or a 5xx every time (#356 review). 

156_CARD_TRIES = 3

159def _rejected(exc): 

160 """Whether Slack refused the card itself, so posting it again will fail the 

161 same way (#356). A SlackApiError carries Slack's answer; a 429 or a 5xx is 

162 Slack being busy or down, so is one of its transient error codes, and 

163 anything that is not a SlackApiError never got an answer at all. Those may 

164 pass, and get the next mention's retry.""" 

165 resp = getattr(exc, "response", None) 

166 status = getattr(resp, "status_code", None) 

167 try: 

168 error = resp.get("error") if resp is not None else None 

169 except Exception: 

170 error = None 

171 retval = (isinstance(exc, SlackApiError) and status is not None and status != 429 

172 and status < 500 and error not in _TRANSIENT_ERRORS) 

173 return retval

176def _card_failed(client, channel, thread_ts, req_id, exc): 

177 """An approval card did not post. Retry it on the next mention when the 

178 failure may pass, up to _CARD_TRIES failures; withdraw it when Slack 

179 rejected the card, or the retries ran out (#356). 

180 

181 A rejected card is rejected again on every mention, and a request no human 

182 can see holds the thread's resume forever. Withdrawn, it never runs, and 

183 the thread says so -- with the id only, since the command is the likeliest 

184 thing the card choked on. 

185 

186 If the thread was only waiting on this one -- another request here was 

187 resolved while it was parked, and that resume was refused -- the thread is 

188 resumed from that outcome now, after the note, so the resumed turn reads 

189 the withdrawal. A thread whose only request was withdrawn is not resumed: 

190 take_deferred() has nothing for it.""" 

191 if not _rejected(exc) and approvals.unsurface(req_id) < _CARD_TRIES: 

192 return 

193 if approvals.withdraw(req_id) is None: 

194 return 

195 try: 

196 client.chat_postMessage( 

197 channel=channel, thread_ts=thread_ts, 

198 text=(f":warning: [{req_id}] needed approval, but its card could not be " 

199 "posted, so it was withdrawn and did not run. Ask again if it is " 

200 "still needed."), 

201 ) 

202 except Exception: 

203 logging.exception("could not report the withdrawn request %s", req_id) 

204 outcome = approvals.take_deferred(channel, thread_ts) 

205 if outcome is not None: 

206 _resume_thread(client, channel, thread_ts, outcome["req_id"], 

207 {"command": outcome["command"], "requester": outcome["requester"]}, 

208 approved=outcome["approved"], result=outcome["result"], 

209 user_id=outcome["user_id"])

218 req["full_link"] = _post_full_command(client, channel, thread_ts, req_id, req["command"])

246def _resolve(ack, body, client, action, run, queue=approvals, claimed=slack_blocks.claimed): 

247 """Shared button plumbing: ack inside Slack's 3s budget, act as the clicking 

248 user (never the model), then rewrite the message so the buttons are gone and 

249 the outcome is on the record.""" 

250 ack() 

251 channel = body["channel"]["id"] 

252 message_ts = body["message"]["ts"] 

253 thread_ts = body["message"].get("thread_ts") or message_ts 

254 ctx = { 

255 "user_id": body["user"]["id"], 

256 "channel": channel, 

257 "thread_ts": thread_ts, 

258 "surface": SlackSurface(client), 

259 } 

260 # A click that resolves nothing must not touch the card (#94). chat_update 

261 # is destructive, and the card is the only place a parked command is ever 

262 # displayed -- overwriting it with a refusal both loses the command text and 

263 # takes the buttons away from the trusted user who could still act on it, 

264 # while the request itself stays in the queue, surfaced and unreachable. 

265 if not admin_tools.is_trusted(ctx["user_id"]): 

266 admin_tools.refuse_click(action["value"], ctx, action.get("action_id")) 

267 return 

268 req_id = action["value"] 

269 # Take the request before touching the card. Two deliveries of one press 

270 # land on two Bolt threads, and the loser -- whose pop finds nothing -- 

271 # would otherwise overwrite the winner's output with "no pending request" 

272 # for a command that did run. Hiding the buttons does not prevent that. 

273 req = queue.acquire(req_id, channel) 

274 if req is None: 

275 # Two reasons acquire can fail, and they deserve different answers. If 

276 # another delivery of this press has it in flight, that thread will 

277 # update the card and saying anything here would claim it is gone while 

278 # it is running -- and it IS gone from the queue by then, since approve 

279 # pops before it executes, so the hold is the only thing that knows. 

280 # Only a genuinely absent request gets a message, and it goes to the 

281 # thread rather than rewriting the card, because a click that resolves 

282 # nothing must not destroy the only copy of the parked command (#94). 

283 if queue.status(req_id, channel)[0] == "held": 

284 return 

285 # Say WHICH kind of gone, because the two need different things from 

286 # the reader (#258). An id from an earlier boot means the process 

287 # restarted under the card: the request is unrecoverable, nobody did 

288 # anything wrong, and asking again is the whole fix. An id from THIS 

289 # boot that is absent was already acted on, or is not this channel's. 

290 stale_boot = not queue.from_this_boot(req_id) 

291 if stale_boot: 

292 note = (f":information_source: [{req_id}] is from before a restart, so that " 

293 "request no longer exists -- ids are per-boot on purpose, so an old " 

294 "card can never resolve to a newer request. Nothing is lost but the " 

295 "card: ask again, and if it still needs approval you get a fresh one.") 

296 else: 

297 note = (f":information_source: [{req_id}] is not pending here -- it was already " 

298 "acted on, or it belongs to another channel.") 

299 try: 

300 client.chat_postMessage(channel=channel, thread_ts=thread_ts, text=note) 

301 except Exception: 

302 logging.exception("could not report a stale approval click") 

303 # ...and retire the card, so a fifth press cannot produce a fifth 

304 # identical line. #94's rule is that a click resolving nothing must not 

305 # destroy the only copy of the parked command -- and this does not: the 

306 # command text stays, only the buttons go. Nobody can act on an absent 

307 # request through them, which is what made the repeat press useless. 

308 try: 

309 client.chat_update( 

310 channel=channel, ts=message_ts, 

311 text=redact.scrub(f"Expired [{req_id}]"), 

312 blocks=slack_blocks.expired(req_id, body.get("message"), stale_boot), 

313 ) 

314 except Exception: 

315 logging.exception("could not retire the stale card for %s", req_id) 

316 return 

317 try: 

318 # A reaction as well as the rewrite (#206). The claimed-card update 

319 # below already says "Working on", but a reaction is the signal people 

320 # are trained on: every user message gets :eyes: the moment the agent 

321 # picks it up, and an approval click got nothing of the kind. Between 

322 # the click and the result -- seconds, on a slow command -- there was no 

323 # way to tell "it landed and is running" from "it was lost", which is 

324 # exactly what the bug #169 fixed used to look like. 

325 slack_tools.react(client, channel, message_ts, add="eyes") 

326 # Acknowledge the click before doing the work (#101), not after. The 

327 # final update can be seconds away -- this one ran a network call -- and 

328 # until it lands the card is unchanged with its buttons still live, 

329 # which reads as a click that went nowhere. 

330 try: 

331 client.chat_update( 

332 channel=channel, 

333 ts=message_ts, 

334 text=f"Working on [{req_id}] for <@{ctx['user_id']}>", 

335 blocks=claimed(action.get("action_id"), req_id, ctx["user_id"], req), 

336 ) 

337 except Exception: 

338 logging.exception("could not mark approval message as claimed") 

339 # The runner gets the request we acquired (#105): it must execute 

340 # what it was handed, never look it up again by id -- the id path 

341 # would find nothing (we hold it) and report "no pending request" 

342 # for a command that is about to run. 

343 result = redact.scrub(run(req_id, req, ctx)) 

344 finally: 

345 queue.release(req_id) 

346 if result.startswith(learning.RETRY) and queue is proposals: 

347 # A transient failure put the proposal back under the same id (#129 

348 # review): the card gets its buttons back, with the reason on it, so 

349 # the trusted user can try again from where they are. 

350 prop = proposals.peek(req_id, channel) 

351 blocks = slack_blocks.proposal(req_id, prop, admin_tools._trusted_tags()) if prop else [] 

352 blocks.insert(0, {"type": "section", "text": {"type": "mrkdwn", "text": f":warning: {result[len(learning.RETRY):].strip()[:2800]}"}}) 

353 else: 

354 blocks = [{"type": "section", "text": {"type": "mrkdwn", "text": f"```{result[:2900]}```"}}] 

355 try: 

356 client.chat_update(channel=channel, ts=message_ts, text=result[:2900], blocks=blocks) 

357 except Exception: 

358 logging.exception("could not update approval message") 

359 client.chat_postMessage(channel=channel, thread_ts=thread_ts, text=result[:2900]) 

360 # The receipt becomes a verdict: :eyes: meant "heard you", and it has to 

361 # stop meaning that once the answer is on the card. Denial is not failure, 

362 # so it gets its own mark rather than the one that reads as an error. 

363 slack_tools.react(client, channel, message_ts, remove="eyes", 

364 add="white_check_mark" if action.get("action_id") == "approve_command" else "no_entry_sign") 

365 if queue is approvals: 

366 _resume_thread(client, channel, thread_ts, req_id, req, 

367 approved=action.get("action_id") == "approve_command", 

368 result=result, user_id=ctx["user_id"])

371def _resume_thread(client, channel, thread_ts, req_id, req, approved, result, user_id): 

372 """Carry the turn on after a click resolved a parked command (#169). 

373 

374 handler.resume decides whether this is the moment -- it returns None while 

375 the thread still has a parked request, so a turn that parked three commands 

376 resumes once, on the last click, rather than three times over. 

377 

378 Best-effort by construction. The click is acked, the command has run and 

379 its output is on the card, so a failure in here costs the continuation and 

380 nothing else; it must never look like the approval failed.""" 

381 try: 

382 context = _thread_context(client, channel, thread_ts, None) 

383 # The starting mention's files, read again (#362): only that one 

384 # message's, and only on a resume, never the whole thread's. 

385 # Fetched only when this click can resume: with other cards still 

386 # parked handler.resume refuses, and the download would be for nothing. 

387 parts, notes = [], [] 

388 if not approvals.pending_in(channel, thread_ts): 

389 try: 

390 save_under, no_save = attachments.save_root(channel) 

391 parts, notes = attachments.to_parts( 

392 _TASK_FILES.get((channel, thread_ts)), 

393 save_under=save_under, no_save=no_save) 

394 except Exception: # noqa: BLE001 -- a bad download must not cost the resume 

395 logging.exception("resume: could not re-read the task's files") 

396 notes = ["the files this task started with could not be read again"] 

397 reply = handler.resume( 

398 req_id, approved, req.get("command", ""), result, 

399 thread_context=context, channel=channel, thread_ts=thread_ts, 

400 # The clicking trusted user, matching the text path: "approve <id>" 

401 # runs inside that user's own turn, so the resumed turn carries the 

402 # same identity a typed approval would. 

403 user_id=user_id, surface=SlackSurface(client), 

404 requester=req.get("requester"), 

405 attachments=parts, attachment_notes=notes, 

406 ) 

407 if reply is None: 

408 # Two different silences used to look alike here, and only one of 

409 # them is worth breaking (#266). The log line said "still parked, 

410 # or already resuming" -- to a file, which is the one reader that 

411 # does not need to know, while the thread showed a command running 

412 # and then nothing, which reads as a task that died. 

413 waiting = approvals.pending_in(channel, thread_ts) 

414 if waiting: 

415 # Read after the fact, so it can be off by one if a click lands 

416 # in between. It is a sentence telling a human roughly how many 

417 # cards to clear, not a control decision -- begin_resume makes 

418 # that one, under a lock. 

419 _plural = "s" if waiting > 1 else "" 

420 try: 

421 client.chat_postMessage( 

422 channel=channel, thread_ts=thread_ts, 

423 text=(f":hourglass: [{req_id}] is resolved, and this task is " 

424 f"waiting on {waiting} more request{_plural} in this " 

425 f"thread. It continues on its own once the last one is " 

426 f"approved or denied -- no need to re-mention me."), 

427 ) 

428 except Exception: 

429 logging.exception("could not report the remaining parked count") 

430 # else: another worker is already resuming this thread; it will 

431 # post the continuation, and a second note would be noise. 

432 return 

433 client.chat_postMessage(channel=channel, thread_ts=thread_ts, **_reply_payload(reply)) 

434 # The resumed turn can park the next step, so its cards need posting 

435 # too -- otherwise the task stops one command later, for the same 

436 # reason it used to stop here. 

437 _post_pending(client, channel, thread_ts) 

438 except Exception: 

439 logging.exception("could not resume the thread after [%s]", req_id)

470def _turn(event, say, client, logger): 

471 """One turn, from whichever Slack event carried it. 

472 

473 A mention in a channel and a direct message are the same turn: the same 

474 handler, the same policy lookup, the same approval cards. Only the door 

475 differs (#23). Keeping one body means a DM cannot quietly miss something a 

476 mention gets -- attachments, thread context, the parked-request cards -- 

477 which is exactly what a second copy of this would drift into.""" 

478 channel = event.get("channel") 

479 # Ack immediately with a reaction so we don't look silent while churning. 

480 # Best-effort: needs reactions:write; if not granted, this no-ops. 

481 try: 

482 client.reactions_add(channel=channel, name="eyes", timestamp=event.get("ts")) 

483 except Exception: 

484 pass 

485 thread_ts = event.get("thread_ts") or event.get("ts") 

486 context = _thread_context(client, channel, thread_ts, event.get("ts")) 

487 # Attachments ride in event["files"], not in the text (#68). Only this 

488 # message's -- files in thread *history* stay unread for now, since every 

489 # reply would re-download them. The refs are kept so a resume after an 

490 # approval card gets this message's files back (#362). Each one is also 

491 # saved under the channel cwd and its path given to the model (#372). 

492 _remember_files(channel, thread_ts, event.get("files")) 

493 save_under, no_save = attachments.save_root(channel) 

494 parts, notes = attachments.to_parts( 

495 event.get("files"), save_under=save_under, no_save=no_save) 

496 text = event.get("text", "") 

497 if notes: 

498 text += "\n\n[attachments I could not read: " + "; ".join(notes) + "]" 

499 # A bot's post carries the bot user's id in `user`. Passed on, it reads as 

500 # an untrusted human and gets the plain-language audience line, which 

501 # changed the reply's shape when a voice listener posted as a bot. A bot is 

502 # not an audience: it gets what a webhook post (no user) gets. 

503 asker = None if event.get("bot_id") else event.get("user") 

504 try: 

505 reply = handler.handle(text, thread_context=context, channel=channel, thread_ts=thread_ts, user_id=asker, surface=SlackSurface(client), attachments=parts) 

506 except Exception as exc: # one clear message, no dozen "did not run" cards 

507 # The exception goes to the log only (#352). A router failure's text is 

508 # the whole multi-provider dump -- deployment hashes, fallback groups, 

509 # vendor JSON with account ids -- which nobody in the channel can act on 

510 # and the operator reads in the log anyway. The log side is scrubbed by 

511 # redact.install_logging(), the traceback included (#72). 

512 logger.exception("handler failed: channel=%s thread=%s", channel, thread_ts) 

513 reply = handler.failure_line(exc) 

514 say(thread_ts=thread_ts, **_reply_payload(reply)) 

515 # Anything the turn parked gets Approve/Deny buttons (#50), so a trusted 

516 # user answers with a click instead of another round-trip through the model. 

517 _post_pending(client, channel, thread_ts) 

518 # A rung found dead during this or an earlier turn (#353), told once, as its 

519 # own post so a mirror never carries it. Only in a trusted user's turn: an 

520 # operator notice belongs to an operator, not a client channel, a stranger's 

521 # DM or a Signal group. Until then it waits in the state file. 

522 if admin_tools.is_trusted(asker): 

523 try: 

524 for notice in llm.claim_dead_notices(): 

525 say(thread_ts=thread_ts, **_reply_payload(notice)) 

526 except Exception: # noqa: BLE001 -- the ERROR log line already has it 

527 logging.exception("could not post a dead-rung notice") 

528 # The question and the answer, into linked gateway conversations that 

529 # mirror this channel (#336). Off unless configured. 

530 gateway_app.mirror_in(channel, event.get("text", ""), reply)

542@app.event("message") 

543def on_message(event, say, client, logger): 

544 """A direct message is a turn, and so is an @here/@channel; every other 

545 channel message still needs a mention. 

546 

547 Trust does not change with the door: `trusted_users` is per user, so the 

548 same person has the same authority in a DM as in a channel, and the channel 

549 id a DM resolves to (`D...`) takes a policy like any other -- one that is 

550 absent falls back to `default_policy`, so a deployment wanting DMs narrower 

551 than its default says so there. The decisions are `identity.dm_turn` and 

552 `identity.broadcast_turn`, where they can be tested without a Slack 

553 connection. 

554 

555 A broadcast is answered because it is addressed to everyone in the room and 

556 the agent is in the room (#260). The case that prompted it: an `@here` 

557 asking why nothing was happening reached every human in the channel and not 

558 the one participant able to answer.""" 

559 if not (identity.dm_turn(event) or identity.broadcast_turn(event)): 

560 return 

561 if _seen(event.get("ts")): 

562 return # a mention inside a DM arrives twice, once per event type 

563 _turn(event, say, client, logger)

633 # Ingest-agnostic: announce knows only how to call post(text).

635 # One dead channel must not silence the rest (#89). A stale or archived 

636 # channel id makes chat_postMessage raise, and an unguarded loop aborts 

637 # before the healthy channels are ever reached -- observed live, where a 

638 # stale DM id sorted first out of the set and the announcement reached

643 # reported failure would re-announce to the channels that already have it 

644 # on every boot -- and the watchdog (#66) makes boots frequent. Fan-out
