1"""Gate a shell command through voitta-yolt's grammar classifier. 

2 

3Subprocesses YOLT's `grammar_classifier.py --no-user-allow '<command>'`, which 

4prints {"decision": "safe"|"unsafe", "reason": ..., "allow_patterns": n}. 

5"safe" == read-only (auto-run); anything else (unsafe, unknown, error, 

6unconfigured) is treated as mutating -> needs approval. Fail-closed: if YOLT 

7can't run, we do NOT auto-run. 

8 

9Every consumer here compares against "safe" rather than against "unsafe", and 

10that is load-bearing rather than stylistic: voitta-yolt 2.0.0 adds a fourth 

11verdict, "deny" (a git-state predicate refusing outright rather than asking). 

12It is the most restrictive thing the classifier can say, so a branch testing 

13`== "unsafe"` would let it fall through to the least restrictive path. 

14 

15`--no-user-allow` is the whole of #148. Without it the classifier promotes any 

16command matching a `Bash(...)` pattern in `~/.claude/settings.json`, 

17`<cwd>/.claude/settings.json` or `<cwd>/.claude/settings.local.json` to "safe" 

18-- and it is subprocessed with this agent's cwd, so the operator's interactive 

19Claude Code permissions and this repo's own settings decided what a Slack 

20channel could auto-run. 123 patterns on the deployment host, among them 

21`gh pr merge*`, `gh api*`, `git push origin feature/*` and `codex exec *`: all 

22mutating, none ever seen by an approval card, all authorized by a file written 

23for a human at a terminal. A confused deputy, and the sandbox does not cover 

24it because these are network effects. 

25 

26So "safe" here means what YOLT's own rules say, and nothing else. A command an 

27operator wants auto-run despite that does not exist yet; when one does, it 

28belongs in shmobster's own config where it is reviewable, not inherited from 

29somewhere else."""

42def _classify_raw(command, flags=(_NO_USER_ALLOW,), cwd=None): 

43 """(dict, error). The dict is YOLT's parsed output; error is a string when 

44 it could not be obtained. 

45 

46 `cwd` is passed as `--cwd`, not as the subprocess's own directory, and the 

47 difference is the whole point (#182). voitta-yolt 2.0.x's `deny` verdict is 

48 produced by git-state predicates -- "would push to the default branch" -- 

49 evaluated against the directory the command would run in. Left unsaid, that 

50 directory is whatever this agent process happens to be in, so the predicate 

51 reads one repository and answers about another: a false deny citing a branch 

52 the channel never named, or no deny at all from a non-git directory. The 

53 flag exists from 2.0.1, and `preflight` refuses the boot without it (#229). 

54 

55 Errors are signalled out of band. On a rejected invocation the classifier 

56 exits non-zero and writes the reason to stderr, leaving stdout empty -- so a 

57 caller that only parses stdout reports a JSON decode error where the real 

58 event was a refusal with a named cause. Check the code, and say what stderr 

59 said.""" 

60 path = config.YOLT_CLASSIFIER 

61 if not path: 

62 return (None, "yolt not configured (exec.yolt_classifier)") 

63 argv = [sys.executable, path, *flags] 

64 if cwd: 

65 argv += ["--cwd", cwd] 

66 argv.append(command) 

67 try: 

68 proc = subprocess.run(argv, capture_output=True, text=True, timeout=15) 

69 except Exception as exc: 

70 return (None, f"yolt error: {exc}") 

71 if proc.returncode != 0: 

72 detail = (proc.stderr or "").strip().splitlines() 

73 why = detail[0] if detail else "no message on stderr" 

74 return (None, f"yolt exited {proc.returncode}: {why}") 

75 try: 

76 return (json.loads(proc.stdout), None) 

77 except Exception as exc: 

78 return (None, f"yolt error: {exc}")

81def classify(command, cwd=None): 

82 data, err = _classify_raw(command, cwd=cwd) 

83 if err: # fail closed -> mutating 

84 retval = ("unsafe", err) 

85 return retval 

86 retval = (data.get("decision", "unsafe"), data.get("reason", "")) 

87 return retval

90# The probe, and the choice matters (#182). It used to be `cat /dev/null`, a 

91# read, asserted to come back `safe` -- which is precisely the assertion #177 

92# removed: under voitta-yolt 2.0.x every ordinary read is `unknown`, delegated 

93# to a host classifier this agent does not have, and the read-only set now lives 

94# in grant.READ_VERBS instead. So the probe stopped asking "do you still call 

95# reads safe" and started asking the question we actually depend on: "do you 

96# accept --cwd". 

97# 

98# It asks with a command no version has ever called anything but unsafe. A 

99# classifier predating 2.0.1 takes its command from the first non-flag argument, 

100# so `--cwd` becomes the command and the verdict is `unknown | no rule: --cwd` -- 

101# a verdict about a flag string, with the `rm` never examined. That is the same 

102# family as handing `--no-user-allow` to a pre-1.2.0 classifier, and it fails 

103# closed and silent: every command in every channel parks, with nothing saying 

104# why. From 2.1.0 the classifier rejects the unknown flag outright and exits 

105# non-zero, which `_classify_raw` now reports with its stderr. 

106_PROBE = "rm -rf /tmp/shmobster-preflight-probe" 

107_PROBE_CWD = "/" 

108_PROBE_ATTEMPTS = 3 

109_PROBE_RETRY_SECONDS = 1.0 

110 

111 

112# The voitta-yolt release this shmobster is tested against. The classifier is 

113# a sibling CLONE, not a package (README, "Upgrading pulls three checkouts"), 

114# so nothing installs a version: it is whatever that clone holds, and a 

115# `git pull` there changes every verdict live, between restarts. On 2026-10-07 

116# another session pulled it from v2.4.0 to v2.7.0 without knowing the agent 

117# read it. Keep the clone detached at this tag; move both together. 

118YOLT_PIN = "v2.17.0"

139def preflight(): 

140 """Reasons this agent must not serve on the YOLT it found (#229). 

141 

142 The caller refuses the boot on any of them. This returns strings rather 

143 than raising so it stays testable offline, which is the same reason 

144 logsetup is its own module. 

145 

146 Three failure modes, all silent, all ending in "every command parks": 

147 

148 An older classifier takes its command from argv[1], so a flag it does not 

149 know lands there and every verdict becomes a verdict about that string. One 

150 that accepts the flags and inherits the allow-lists anyway is worse -- the 

151 old auto-run surface with nothing to show for it, which is why the count has 

152 to be reported and has to be zero. 

153 

154 And one that does not accept `--cwd`, which this agent depends on for a 

155 reason no verdict reveals: without it voitta-yolt 2.0.x's git-state 

156 predicates judge whichever directory this process is in rather than the 

157 channel's, and a `deny` layer asked about the wrong repository does not 

158 error or warn, it simply never denies (#182).""" 

159 retval = [] 

160 # Retried before it is believed (#229). Since this refuses the boot rather 

161 # than logging, a momentary failure -- a slow fork, a machine still waking 

162 # -- would become a launchd crash loop, which is how #230's 251 MB log 

163 # happened. Three attempts, and only a persistent failure counts. 

164 for _attempt in range(_PROBE_ATTEMPTS): 

165 data, err = _classify_raw(_PROBE, cwd=_PROBE_CWD) 

166 if not err: 

167 break 

168 if _attempt + 1 < _PROBE_ATTEMPTS: 

169 time.sleep(_PROBE_RETRY_SECONDS) 

170 if err: 

171 retval.append(err) 

172 return retval 

173 if data.get("decision") != "unsafe": 

174 retval.append( 

175 f"yolt called {_PROBE!r} {data.get('decision')!r} rather than unsafe when asked " 

176 f"with --cwd, so it is classifying the flag instead of the command and every " 

177 f"command in every channel will park. This yolt predates --cwd (2.0.1+); " 

178 f"upgrade it" 

179 ) 

180 elif "allow_patterns" not in data: 

181 retval.append( 

182 f"yolt does not report allow_patterns, so {_NO_USER_ALLOW} cannot be confirmed; " 

183 "the auto-run set may still be inherited from ~/.claude/settings.json. " 

184 "Upgrade voitta-yolt to 1.2.0+" 

185 ) 

186 elif data["allow_patterns"]: 

187 retval.append( 

188 f"yolt loaded {data['allow_patterns']} user allow pattern(s) despite " 

189 f"{_NO_USER_ALLOW}; the auto-run set is wider than YOLT's own rules" 

190 ) 

191 else: 

192 logging.info("yolt: 0 inherited allow patterns; 'safe' means YOLT's own rules (#148)") 

193 return retval

196def is_read_only(command, cwd=None): 

197 decision, _ = classify(command, cwd=cwd) 

198 retval = decision == "safe" 

199 return retval
